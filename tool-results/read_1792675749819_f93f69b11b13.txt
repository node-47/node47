   232→   232→function isSectionTitle(text: string): boolean {
   233→   233→  const t = stripAccents(text)
   234→   234→    .toLowerCase()
   235→   235→    .replace(/[^a-z& ]+/g, " ")
   236→   236→    .replace(/\s+/g, " ")

   241→   241→
   242→   242→/** Découpe le texte en sections nommées (A à H). Accepte :
   243→   243→ *  - lettres « A. / B) / C - » (majuscules ou minuscules, avec ou sans gras)
   244→   244→ *  - numérotation quelconque « 1. 2. » « I. II. » si le titre est reconnu
   245→   245→ *  - titres nus « MATCH ET CONTEXTE » sans aucun préfixe
   246→   246→ *  Sans en-têtes reconnaissables → tout dans "ALL" (remplissage par étiquettes). */
   247→   247→function splitSections(text: string): Record<string, string> {
   248→   248→  const lines = cleanText(text).split("\n");
   249→   249→  const sections: Record<string, string[]> = { ALL: [] };
   250→   250→  let current = "ALL";
   251→   251→  const headerRe = /^\s*([A-Ha-h])\s*[.)\]:\-–—]\s*(.+?)\s*$/;
   252→   252→  const bulletRe = /^\s*[•‣▪◦*\-–—]\s+/;
   253→   253→  for (const line of lines) {
   254→   254→    const stripped = stripInline(line);
   255→   255→    if (!stripped || DECOR_LINE_RE.test(stripped)) continue; // vide / décoratif
   256→   256→    const m = stripped.match(headerRe);
   257→   257→    const letters = m ? m[2].replace(/[^A-Za-zÀ-ÿ]/g, "") : "";
   258→   258→    const isHeader =
   259→   259→      !!m && letters.length > 2 && (letters === letters.toUpperCase() || isSectionTitle(m[2]));
   260→   260→    if (isHeader && m) {
   261→   261→      current = m[1].toUpperCase();
   262→   262→      if (!sections[current]) sections[current] = [];
   263→   263→      continue;
   264→   264→    }
   265→   265→    // En-tête sans lettre reconnue (numéroté ou nu) : titre court, pas de puce,
   266→   266→    // pas de « : » — la position dans l'ordre canonique tranche la section.
   267→   267→    // Pour une ligne NUMÉROTÉE (« 2. Convergence… »), le titre doit rester très
   268→   268→    // court : au-delà, c'est presque toujours une raison/liste, pas un titre.
   269→   269→    const numberedRe = /^\s*(?:\d{1,2}|[ivxlIVXL]{1,5})\s*[.)\]:\-–—]/;
   270→   270→    const maxTitle = numberedRe.test(stripped) ? 45 : 80;
   271→   271→    if (!bulletRe.test(stripped) && !/:/.test(stripped) && stripped.length < maxTitle) {
   272→   272→      const t = stripAccents(stripped)
   273→   273→        .toLowerCase()
   274→   274→        .replace(/^\s*(?:\d{1,2}|[ivxlIVXL]{1,5})\s*[.)\]:\-–—]?\s*/, "")
   275→   275→        .replace(/\s+/g, " ")
   276→   276→        .trim();
   277→   277→      const hit = ORDER_KEYWORDS.find(([re]) => re.test(t));
   278→   278→      if (hit) {
   279→   279→        current = hit[1];
   280→   280→        if (!sections[current]) sections[current] = [];
   281→   281→        continue;
   282→   282→      }
   283→   283→    }
   284→   284→    sections[current].push(stripped);
   285→   285→  }
   286→   286→  const out: Record<string, string> = {};
   287→   287→  for (const [k, v] of Object.entries(sections)) out[k] = v.join("\n").trim();
   288→   288→  return out;
   289→   289→}

   291→   291→/** Découpe une section en puces « Label : corps » (•, -, –, —, *) ; les lignes
   292→   292→    simples sont rattachées à la puce précédente (continuation / sous-listes). */
   293→   293→function parseBullets(section: string): { label: string; body: string }[] {
   294→   294→  const lines = cleanText(section).split("\n");
   295→   295→  const bullets: { label: string; body: string }[] = [];
   296→   296→  const bulletRe = /^\s*[•‣▪◦*\-–—]\s+(.*)$/;
   297→   297→  for (const line of lines) {
   298→   298→    if (DECOR_LINE_RE.test(line.trim())) continue; // séparateur décoratif
   299→   299→    const m = line.match(bulletRe);
   300→   300→    if (m) {
   301→   301→      const t = m[1].trim();
   302→   302→      const ci = t.indexOf(":");
   303→   303→      if (ci > 0 && ci < 80) {
   304→   304→        bullets.push({ label: t.slice(0, ci).trim(), body: t.slice(ci + 1).trim() });
   305→   305→      } else {
   306→   306→        bullets.push({ label: "", body: t });
   307→   307→      }
   308→   308→    } else if (line.trim()) {
   309→   309→      const t = line.trim();
   310→   310→      // Ligne « Label : valeur » SANS puce (collage sans mise en forme) :
   311→   311→      // traitée comme une puce étiquetée plutôt que comme du prose perdue.
   312→   312→      const lm = t.match(LABEL_LINE_RE);
   313→   313→      if (lm) {
   314→   314→        bullets.push({ label: lm[1].trim(), body: lm[2].replace(STOP_LABEL_RE, "").trim() });
   315→   315→      } else if (bullets.length) {
   316→   316→        bullets[bullets.length - 1].body += `\n${t}`;
   317→   317→      } else {
   318→   318→        bullets.push({ label: "", body: t });
   319→   319→      }
   320→   320→    }
   321→   321→  }
   322→   322→  return bullets;
   323→   323→}

   342→   342→  if (!flat) return [];

   344→   344→    .split(/\s*;\s*/)

   346→   346→    .filter((s) => s.replace(/\W/g, "").length >= 12);
   347→   347→  if (semi.length > 1) return semi;
   348→   348→  if (flat.length <= 220) return [flat];

   350→   350→    .split(/(?<=[.!?])\s+/)
