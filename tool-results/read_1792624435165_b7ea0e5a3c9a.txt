   207→function isSectionTitle(text: string): boolean {
   208→  const t = stripAccents(text)
   209→    .toLowerCase()
   210→    .replace(/[^a-z& ]+/g, " ")
   211→    .replace(/\s+/g, " ")
   212→    .trim();
   213→  return SECTION_KEYWORDS.some((k) => t.startsWith(k));
   214→}
   215→
   216→/** Découpe le texte en sections nommées (A à H). Accepte :
   217→ *  - lettres « A. / B) / C - » (majuscules ou minuscules, avec ou sans gras)
   218→ *  - numérotation quelconque « 1. 2. » « I. II. » si le titre est reconnu
   219→ *  - titres nus « MATCH ET CONTEXTE » sans aucun préfixe
   220→ *  Sans en-têtes reconnaissables → tout dans "ALL" (remplissage par étiquettes). */
   221→function splitSections(text: string): Record<string, string> {
   222→  const lines = cleanText(text).split("\n");
   223→  const sections: Record<string, string[]> = { ALL: [] };
   224→  let current = "ALL";
   225→  const headerRe = /^\s*([A-Ha-h])\s*[.)\]:\-–—]\s*(.+?)\s*$/;
   226→  const bulletRe = /^\s*[•‣▪◦*\-–—]\s+/;
   227→  for (const line of lines) {
   228→    const stripped = stripInline(line);
   229→    const m = stripped.match(headerRe);
   230→    const letters = m ? m[2].replace(/[^A-Za-zÀ-ÿ]/g, "") : "";
   231→    const isHeader =
   232→      !!m && letters.length > 2 && (letters === letters.toUpperCase() || isSectionTitle(m[2]));
   233→    if (isHeader && m) {
   234→      current = m[1].toUpperCase();
   235→      if (!sections[current]) sections[current] = [];
   236→      continue;
   237→    }
   238→    // En-tête sans lettre reconnue (numéroté ou nu) : titre court, pas de puce,
   239→    // pas de « : » — la position dans l'ordre canonique tranche la section.
   240→    if (!bulletRe.test(stripped) && !/:/.test(stripped) && stripped.length < 80) {
   241→      const t = stripAccents(stripped)
   242→        .toLowerCase()
   243→        .replace(/^\s*(?:\d{1,2}|[ivxlIVXL]{1,5})\s*[.)\]:\-–—]?\s*/, "")
   244→        .replace(/\s+/g, " ")
   245→        .trim();
   246→      const hit = ORDER_KEYWORDS.find(([re]) => re.test(t));
   247→      if (hit) {
   248→        current = hit[1];
   249→        if (!sections[current]) sections[current] = [];
   250→        continue;
   251→      }
   252→    }
   253→    sections[current].push(stripped);
   254→  }
   255→  const out: Record<string, string> = {};
   256→  for (const [k, v] of Object.entries(sections)) out[k] = v.join("\n").trim();
   257→  return out;
   258→}

   260→/** Découpe une section en puces « Label : corps » (•, -, –, —, *) ; les lignes
   261→    simples sont rattachées à la puce précédente (continuation / sous-listes). */
   262→function parseBullets(section: string): { label: string; body: string }[] {
   263→  const lines = cleanText(section).split("\n");
   264→  const bullets: { label: string; body: string }[] = [];
   265→  const bulletRe = /^\s*[•‣▪◦*\-–—]\s+(.*)$/;
   266→  for (const line of lines) {
   267→    const m = line.match(bulletRe);
   268→    if (m) {
   269→      const t = m[1].trim();
   270→      const ci = t.indexOf(":");
   271→      if (ci > 0 && ci < 80) {
   272→        bullets.push({ label: t.slice(0, ci).trim(), body: t.slice(ci + 1).trim() });
   273→      } else {
   274→        bullets.push({ label: "", body: t });
   275→      }
   276→    } else if (line.trim()) {
   277→      const t = line.trim();
   278→      // Ligne « Label : valeur » SANS puce (collage sans mise en forme) :
   279→      // traitée comme une puce étiquetée plutôt que comme du prose perdue.
   280→      const lm = t.match(LABEL_LINE_RE);
   281→      if (lm) {
   282→        bullets.push({ label: lm[1].trim(), body: lm[2].replace(STOP_LABEL_RE, "").trim() });
   283→      } else if (bullets.length) {
   284→        bullets[bullets.length - 1].body += `\n${t}`;
   285→      } else {
   286→        bullets.push({ label: "", body: t });
   287→      }
   288→    }
   289→  }
   290→  return bullets;
   291→}

   301→/** Items d'une liste de raisons : sous-puces / lignes numérotées, sinon phrases. */
   302→function splitReasons(body: string): string[] {
   303→  const lines = body
   304→    .split("\n")
   305→    .map((l) => l.replace(/^\s*[-–—•*]\s*/, "").replace(/^\d+\s*[.)]\s*/, "").trim())
   306→    .filter(Boolean);
   307→  if (lines.length > 1) return lines;
   308→  const flat = body.replace(/\s+/g, " ").trim();
   309→  if (!flat) return [];
   310→  if (flat.length <= 220) return [flat];
   311→  const sentences = flat
   312→    .split(/(?<=[.!?])\s+/)
   313→    .map((s) => s.trim())
   314→    .filter((s) => s.length >= 24);
   315→  return sentences.length > 1 ? sentences : [flat];
   316→}

   329→function extractCompetition(body: string): { competition?: string; sportHint?: string } {
   330→  const flat = stripAccents(body).toLowerCase().replace(/\s+/g, " ");
   331→  for (const c of COMPETITIONS) {
   332→    if (c.re.test(flat)) return { competition: c.label, sportHint: c.sport };
   333→  }
   334→  // Repli : partie après le séparateur « / » puis « – » / « — » / « - »
   335→  let tail = "";
   336→  const slashParts = body.split(/\s*\/\s*/);
   337→  if (slashParts.length > 1) tail = slashParts.slice(1).join(" / ");
   338→  if (!tail) {
   339→    const dashParts = body.split(/\s+[–—-]\s+/);
   340→    if (dashParts.length > 1) tail = dashParts.slice(1).join(" — ");
   341→  }
   342→  if (!tail) return {};
   343→  tail = cleanCompetitionLabel(tail);
   344→  if (tail.length >= 3 && tail.length <= 80) return { competition: tail };
   345→  return {};
   346→}

   350→function parseMatchDate(body: string, warnings: string[]): string | undefined {
   351→  const flat = body.replace(/\n/g, " ");
   352→  const monthKeys = Object.keys(MONTHS).sort((a, b) => b.length - a.length).join("|");
   353→
   354→  // Format littéraire : « 17 octobre 2026 » / « 3 novembre 2025 »
   355→  let day: number | undefined;
   356→  let month: number | undefined;
   357→  let year: number | undefined;
   358→  const dm = flat.match(new RegExp(`(\\d{1,2})(?:er)?\\s+(${monthKeys})\\s+(\\d{4})`, "i"));
   359→  if (dm) {
   360→    day = parseInt(dm[1], 10);
   361→    month = MONTHS[dm[2].toLowerCase()];
   362→    year = parseInt(dm[3], 10);
   363→  } else {
   364→    // Formats numériques : 17/10/2026, 17-10-2026, 17.10.2026, 2026-10-17
   365→    const ymd = flat.match(/\b(\d{4})[-/](\d{1,2})[-/](\d{1,2})\b/);
   366→    const dmy = ymd ? undefined : flat.match(/\b(\d{1,2})[/.‐-](\d{1,2})[/.‐-](\d{4})\b/);
   367→    const parts = ymd ?? dmy;
   368→    if (parts) {
   369→      const a = parseInt(parts[1], 10);
   370→      const b = parseInt(parts[2], 10);
   371→      const c = parseInt(parts[3], 10);
   372→      if (ymd) {
   373→        year = a; month = b; day = c;
   374→      } else {
   375→        // Ambiguïté jour/mois : la valeur > 12 désigne forcément le jour
   376→        day = a > 12 ? a : b > 12 ? b : a;
   377→        month = day === a ? b : a;
   378→        year = c;
   379→      }
   380→    }
   381→  }
   382→  if (day === undefined || month === undefined || year === undefined) return undefined;
   383→  if (month < 1 || month > 12 || day < 1 || day > 31) return undefined;
   384→
   385→  let hour = 12;
   386→  let minute = 0;
   387→  const tm = flat.match(/(\d{1,2})\s*[h:]\s*(\d{2})?/i);
   388→  let hasTime = false;
   389→  if (tm) {
   390→    hour = parseInt(tm[1], 10);
   391→    minute = tm[2] ? parseInt(tm[2], 10) : 0;
   392→    hasTime = true;
   393→  }
   394→
   395→  const tzm = flat.match(
   396→    /\b(CEST|CETE?|WEST|WET|EEST|EET|PDT|PST|MDT|MST|CDT|CST|EDT|EST|UTC|GMT)\b/i
   397→  );
   398→  let offset = 0;
   399→  if (tzm) {
   400→    offset = TZ_OFFSETS[tzm[1].toUpperCase()] ?? 0;
   401→  } else {
   402→    warnings.push("fuseau horaire non détecté — heure lue comme UTC, vérifiez la date");
   403→  }
   404→  if (!hasTime) {
   405→    warnings.push("heure du match non détectée — 12 h placée par défaut");
   406→  }
   407→
   408→  const utcMs = Date.UTC(year, month - 1, day, hour, minute) - offset * 3_600_000;
   409→  return toDatetimeLocal(new Date(utcMs));
   410→}

   412→/* ---------- Statistiques clés (section C) ---------- */
   413→
   414→/**
   415→ * Extraction prudente : seules les statistiques « propres » sont conservées
   416→ * (étiquette courte et lisible devant le nombre, ou étiquette après « : »).
   417→ * Les fragments de prose sont ignorés (comptés dans `skipped`) afin de ne pas
   418→ * polluer le public — le texte brut reste disponible dans le scénario.
   419→ */
   420→function parseKeyStats(body: string): { stats: KeyStatItem[]; skipped: number } {
   421→  // Protège les décimales (2,33) avant découpe sur les virgules
   422→  const flat = body.replace(/\s+/g, " ").replace(/(\d),(\d)/g, "$1\u0001$2");
   423→  const clauses = flat
   424→    .split(/\s*[;.]\s+|\s*[;]\s*|\s*,\s+/)
   425→    .map((c) => c.replace(/\u0001/g, ",").trim())
   426→    .filter((c) => stripAccents(c).length >= 6);
   427→
   428→  const stats: KeyStatItem[] = [];
   429→  let skipped = 0;
   430→  for (const clause of clauses) {
   431→    if (stats.length >= MAX_POINTS) break;
   432→    const ci = clause.indexOf(":");
   433→    const base = ci > 0 ? clause.slice(ci + 1) : clause;
   434→
   435→    const numRe = /([+\-−~]?\s?\d[\d\u0001\u202f ]*(?:[.,]\d+)?)/;
   436→    const nm = base.match(numRe);
   437→    if (!nm) continue;
   438→    const after = base.slice((nm.index ?? 0) + nm[0].length).trim();
   439→    const unit = after.match(/^(%|u\b|buts?|jeux?|points?|pts?\b|match(?:e)?s?|set|sets|aces?|fautes?)/i);
   440→    const unitTxt = unit ? unit[0] : "";
   441→    const numStr = nm[0].replace(/\u0001/g, ",").replace(/\s{2,}/g, " ").trim();
   442→
   443→    // Année isolée (1900-2099) sans unité → pas une statistique
   444→    if (!unitTxt && /^(19|20)\d{2}$/.test(numStr)) {
   445→      skipped++;
   446→      continue;
   447→    }
   448→
   449→    let label: string;
   450→    if (ci > 0) {
   451→      label = clause.slice(0, ci).trim();
   452→      if (label.length > 64 || !/[A-Za-zÀ-ÿ]/.test(label)) {
   453→        skipped++;
   454→        continue;
   455→      }
   456→    } else {
   457→      const before = base.slice(0, nm.index ?? 0).trim();
   458→      if (before.length < 2 || before.length > 48 || before.includes("(") || !/[A-Za-zÀ-ÿ]/.test(before)) {
   459→        skipped++;
   460→        continue;
   461→      }
   462→      label = before;
   463→    }
   464→    label = label
   465→      .replace(/[\s:;,–—-]+$/, "")
   466→      .replace(/\s+(dans|de|du|des|la|le|les|un|une|en|sur|par|pour|avec|et|ou|au|aux)\s*$/i, "")
   467→      .replace(/[\s:;,–—-]+$/, "")
   468→      .trim();
   469→    if (label.length < 2) {
   470→      skipped++;
   471→      continue;
   472→    }
   473→
   474→    const detail = (unitTxt ? after.slice(unitTxt.length) : after).replace(/^[\s,%–—-]+/, "").trim();
   475→    stats.push({
   476→      label: label.slice(0, 120),
   477→      value: (numStr + (unitTxt ? ` ${unitTxt}` : "")).slice(0, 60),
   478→      ...(detail && detail !== label ? { detail: detail.slice(0, 300) } : {}),
   479→    });
   480→  }
   481→  return { stats, skipped };
   482→}

   484→/* ---------- Niveau de confiance ---------- */

   486→function levelToConfidence(level: string): string {
   487→  if (level.startsWith("tr")) return "TRES_ELEVE";
   488→  if (level.startsWith("faible")) return "FAIBLE";
   489→  if (level.startsWith("modere")) return "MODERE";
   490→  return "ELEVE";
   491→}

   493→const CONF_LEVEL = "(?:tr\\s?e\\s?s\\s+)?(?:eleve|faible|modere)";

   495→/* ---------- Parseur principal ---------- */
   496→
   497→export function parseAnalyzeText(raw: string): ImportReport {
   498→  const warnings: string[] = [];
   499→  const parseErrors: string[] = [];
   500→  const values: ImportedValues = {};
   501→  const filled: string[] = [];
   502→  const synthesisBlocks: string[] = [];
   503→  let contextLine = "";
   504→  // Texte complet aplati + détecteur de puces (pour le repli global)
   505→  const flatRawAll = cleanText(raw).replace(/\s+/g, " ");
   506→  const bulletReTop = /^\s*[•‣▪◦*\-–—]\s+/;
   507→
   508→  if (!raw || stripAccents(raw).replace(/\W/g, "").length < 40) {
   509→    return { values, filled, warnings, parseErrors: ["Texte trop court ou vide."] };
   510→  }
   511→
   512→  const sections = splitSections(raw);
   513→  const hasSections = Object.keys(sections).some((k) => k !== "ALL");
   514→  if (!hasSections) {
   515→    warnings.push("sections A à G non repérées — remplissage partiel par étiquettes");
   516→  }
   517→  const sec = (k: string) => sections[k] ?? (hasSections ? "" : sections.ALL ?? "");
   518→  const bulletsOf = (k: string) => parseBullets(sec(k));
   519→
   520→  /* ---- Titre (secours pour l'affiche) ---- */
   521→  const titleLine = cleanText(raw).match(/Analyse\s*:\s*(.+)$/im);
   522→  if (titleLine) {
   523→    const title = titleLine[1].trim();
   524→    if (title.length >= 3 && title.length <= 160) {
   525→      values.matchName = title;
   526→      filled.push("affiche (titre)");
   527→    }
   528→  }
   529→
   530→  /* ---- A. Match et contexte ---- */
   531→  const aBullets = bulletsOf("A");
   532→  if (aBullets.length) {
   533→    const matchB = findBullet(aBullets, /^(match|rencontre|affiche|evenement)\b/);
   534→    if (matchB?.body) {
   535→      values.matchName = matchB.body.replace(/\s+/g, " ").trim().replace(/[.]\s*$/, "").slice(0, 160);
   536→      filled.push("affiche");
   537→    }
   538→    const sportB = findBullet(aBullets, /^sport/);
   539→    const compB = findBullet(aBullets, /^competition/);
   540→    // La compétition peut être collée à la ligne « Sport » OU avoir sa propre ligne :
   541→    // les deux sources sont examinées, indépendamment l'une de l'autre.
   542→    const compSource = sportB?.body || compB?.body || "";
   543→    if (sportB?.body || compB?.body) {
   544→      const flat = stripAccents(`${sportB?.body ?? ""} ${compB?.body ?? ""}`).toLowerCase();
   545→      if (/\bfootball\b|\bsoccer\b|\bfoot\b/.test(flat)) values.sport = "FOOTBALL";
   546→      else if (/\bbasket/.test(flat)) values.sport = "BASKETBALL";
   547→      else if (/\btennis\b/.test(flat)) values.sport = "TENNIS";
   548→
   549→      const comp =
   550→        (compB?.body ? extractCompetition(compB.body) : {});
   551→      // Repli sur la ligne « Sport » si la ligne « Compétition » n'a rien donné
   552→      if (!comp.competition && sportB?.body) {
   553→        const c2 = extractCompetition(sportB.body);
   554→        if (c2.competition) (comp as { competition?: string }).competition = c2.competition;
   555→        if (c2.sportHint && !comp.sportHint) (comp as { sportHint?: string }).sportHint = c2.sportHint;
   556→      }
   557→      if (comp.sportHint && !values.sport) values.sport = comp.sportHint;
   558→      if (values.sport) filled.push("sport");
   559→      else warnings.push("sport non reconnu — laissez la valeur actuelle");
   560→
   561→      if (comp.competition) {
   562→        values.competition = comp.competition.slice(0, 120);
   563→        filled.push("compétition");
   564→      } else if (compB?.body) {
   565→        const label = cleanCompetitionLabel(compB.body);
   566→        if (label) {
   567→          values.competition = label.slice(0, 120);
   568→          filled.push("compétition");
   569→        }
   570→      }
   571→      if (!values.competition) {
   572→        warnings.push("compétition introuvable dans la section A — à compléter");
   573→      }
   574→    }
   575→    const dateB = findBullet(aBullets, /^date/);
   576→    if (dateB?.body) {
   577→      const local = parseMatchDate(dateB.body, warnings);
   578→      if (local) {
   579→        values.matchDate = local;
   580→        filled.push("date du match");
   581→        warnings.push("date/heure importée — à contrôler");
   582→        const monthKeys2 = Object.keys(MONTHS).sort((a, b) => b.length - a.length).join("|");
   583→        const dateRe2 = new RegExp(`^\\d{1,2}(?:er)?\\s+(${monthKeys2})\\s+\\d{4}$`, "i");
   584→        const weekdayRe = /^(lundi|mardi|mercredi|jeudi|vendredi|samedi|dimanche)$/i;
   585→        const timeRe = /^\d{1,2}\s*[h:]\s*\d{0,2}/i;
   586→        // Retire les parenthèses contenant une heure (ex. « (autour de 08:30–10:30 …) »)
   587→        const cleaned = dateB.body.replace(/\([^()]*\d{1,2}\s*[h:][^()]*\)/g, " ");
   588→        const rest = cleaned
   589→          .split(",")
   590→          .map((s) => s.trim())
   591→          .filter((s) => {
   592→            if (!s) return false;
   593→            const t = s.replace(/\s+/g, " ").trim();
   594→            const stripped = t.replace(/^(lundi|mardi|mercredi|jeudi|vendredi|samedi|dimanche)\s+/i, "");
   595→            if (weekdayRe.test(t)) return false;
   596→            if (dateRe2.test(stripped)) return false;
   597→            if (timeRe.test(t) && t.length < 24) return false;
   598→            return true;
   599→          })
   600→          .join(", ")
   601→          .replace(/\s{2,}/g, " ")
   602→          .trim();
   603→        if (rest && rest.length > 8) contextLine = rest.slice(0, 300);
   604→      }
   605→    } else {
   606→      warnings.push("date du match introuvable dans la section A");
   607→    }
   608→  }
   609→
   610→  /* ---- B. Scénario sportif ---- */
   611→  const bBullets = bulletsOf("B");
   612→  const keyPointItems: string[] = [];
   613→  const summaryParts: string[] = [];
   614→
   615→  if (bBullets.length) {
   616→    for (const b of bBullets) {
   617→      const lab = normLabel(b.label);
   618→      if (/^raisons principales/.test(lab)) {
   619→        keyPointItems.push(...splitReasons(b.body));
   620→      } else if (/^scenario alternatif|^alternatif/.test(lab)) {
   621→        if (b.body) {
   622→          values.alternativeScenario = b.body.replace(/\s+/g, " ").trim().slice(0, 2000);
   623→          filled.push("scénario alternatif");
   624→        }
   625→      } else if (/^facteurs/.test(lab)) {
   626→        const items = b.body
   627→          .split(/\s*;\s*|\n/)
   628→          .map((l) => l.replace(/^\s*[-–—•*]\s*/, "").replace(/^\d+\s*[.)]\s*/, "").trim())
   629→          .filter((l) => l.length >= 6);
   630→        keyPointItems.push(...items);
   631→      } else if (!b.label && b.body && b.body.replace(/\W/g, "").length >= 12) {
   632→        // Sous-puces imbriqués (ex. liste de raisons sans étiquette)
   633→        const items = b.body
   634→          .split("\n")
   635→          .map((l) => l.replace(/^\s*[-–—•*]\s*/, "").replace(/^\d+\s*[.)]\s*/, "").trim())
   636→          .filter((l) => l.length >= 6);
   637→        keyPointItems.push(...items);
   638→      } else if (b.label) {
   639→        summaryParts.push(`• ${b.label} : ${b.body.replace(/\s+/g, " ").trim()}`);
   640→      } else {
   641→        summaryParts.push(b.body.trim());
   642→      }
   643→    }
   644→  }
   645→
   646→  /* ---- C. Convergence des preuves ---- */
   647→  const cBullets = bulletsOf("C");
   648→  const convergenceParts: string[] = [];
   649→  if (cBullets.length) {
   650→    for (const b of cBullets) {
   651→      const lab = normLabel(b.label);
   652→      if (/^statistiques?/.test(lab) && b.body) {
   653→        const { stats, skipped } = parseKeyStats(b.body);
   654→        if (stats.length) {
   655→          values.keyStats = stats;
   656→          filled.push(`statistiques clés (${stats.length})`);
   657→          if (skipped > 0) {
   658→            warnings.push(`statistiques partielles (${skipped} fragment(s) non décodé(s)) — vérifiez la section C`);
   659→          }
   660→        } else {
   661→          if (skipped > 0) {
   662→            warnings.push("statistiques non décodées — section C à saisir à la main");
   663→          }
   664→          convergenceParts.push(`• ${b.label} : ${b.body.replace(/\s+/g, " ").trim()}`);
   665→        }
   666→      } else if (b.label) {
   667→        convergenceParts.push(`• ${b.label} : ${b.body.replace(/\s+/g, " ").trim()}`);
   668→      } else {
   669→        convergenceParts.push(b.body.trim());
   670→      }
   671→    }
   672→  }
   673→
   674→  /* ---- D. Comparaison des marchés ---- */
   675→  const dText = sec("D").trim();
   676→
   677→  /* ---- Assemblage du scénario principal ----
   678→     Sections reconnues : assemblage B + convergence + comparaison.
   679→     Sans sections (collage plat) : texte narratif = tout SAUF les lignes
   680→     « Label : valeur » de champs (déjà versés dans le formulaire) et la
   681→     ligne d'affiche (déjà dans matchName). */
   682→  const vsLineRe = /^[A-Za-zÀ-ÿ0-9.'\'\- ]{3,48}\s+vs\.?\s+[A-Za-zÀ-ÿ0-9.'\'\- ]{3,48}\s*$/;
   683→  if (!hasSections) {
   684→    const fieldLabelsNoNarr = FLAT_LABELS.filter((l) => l !== "equipe superieure");
   685→    const fieldLineRe = new RegExp(
   686→      `^(?:[-–—•*]\\s*)?(?:${fieldLabelsNoNarr.map((l) => accentPat(l).replace(/ /g, "\\s+")).join("|")})\\s*[:\\-–—]`,
   687→      "i"
   688→    );
   689→    const narrative = cleanText(raw)
   690→      .split("\n")
   691→      .map((l) => stripInline(l).trim())
   692→      .filter(Boolean)
   693→      .filter((l) => !fieldLineRe.test(l))
   694→      .filter((l) => !vsLineRe.test(l));
   695→    if (narrative.length) {
   696→      values.scenarioSummary = capText(narrative.join("\n"), MAX_SUMMARY, warnings, "scénario");
   697→      filled.push("scénario sportif");
   698→    }
   699→  } else {
   700→    const summaryChunks: string[] = [];
   701→    if (contextLine) summaryChunks.push(contextLine);
   702→    summaryChunks.push(...summaryParts);
   703→    if (convergenceParts.length) {
   704→      summaryChunks.push("Convergence des preuves :", ...convergenceParts);
   705→    }
   706→    if (dText) summaryChunks.push("Comparaison des marchés :", dText);
   707→    const summary = summaryChunks.join("\n").trim();
   708→    if (summary) {
   709→      values.scenarioSummary = capText(summary, MAX_SUMMARY, warnings, "scénario");
   710→      filled.push("scénario sportif");
   711→    }
   712→  }
   713→
   714→  if (keyPointItems.length) {
   715→    const cleaned = keyPointItems.filter((k) => !vsLineRe.test(k.trim()));
   716→    if (cleaned.length) {
   717→      const capped = cleaned.slice(0, MAX_POINTS).map((k) => k.slice(0, 480));
   718→      values.keyPoints = capped.join("\n");
   719→      filled.push(`points clés (${capped.length})`);
   720→      if (cleaned.length > MAX_POINTS) {
   721→        warnings.push(`points clés limités à ${MAX_POINTS} — ${cleaned.length - MAX_POINTS} ignorés`);
   722→      }
   723→    }
   724→  }
   725→
   726→  /* ---- E. Marché retenu ---- */
   727→  const eBullets = bulletsOf("E");
   728→  if (eBullets.length) {
   729→    const marketB = findBullet(eBullets, /^(marche|market|marche\s+retenu|pari\s+retenu)\b/);
   730→    if (marketB?.body) {
   731→      let mk = marketB.body.replace(/\s+/g, " ").trim();
   732→      mk = mk.split(/\s+\(ou\b/i)[0].trim(); // retire le second choix éventuel
   733→      mk = mk.replace(/[;.。]+\s*$/, "").trim();
   734→      values.market = mk.slice(0, 120);
   735→      filled.push("marché");
   736→    }
   737→    const selB = findBullet(eBullets, /^(selection|pronostic|pari|choix|pick|pronostic\s+retenu)\b/);
   738→    if (selB?.body) {
   739→      values.selection = selB.body
   740→        .replace(/\s+/g, " ")
   741→        .trim()
   742→        .replace(/[;.。]+\s*$/, "")
   743→        .slice(0, 120);
   744→      filled.push("sélection");
   745→    }
   746→    const lineB = findBullet(eBullets, /^ligne\b/);
   747→    if (lineB?.body) {
   748→      const lm = lineB.body.match(/^\s*([+\-−–—]?\s?\d+(?:[.,]\d+)?)/);
   749→      if (lm) {
   750→        values.line = lm[1].replace(/[−–—]/g, "-").replace(/\s+/g, "").slice(0, 60);
   751→        filled.push("ligne");
   752→      }
   753→    }
   754→    const oddsB = findBullet(eBullets, /^cote/);
   755→    if (oddsB?.body) {
   756→      const dec = oddsB.body.match(/\d{1,2}[.,]\d{1,3}/);
   757→      let oddsVal: string | undefined;
   758→      if (dec) {
   759→        oddsVal = dec[0].replace(",", ".");
   760→      } else {
   761→        const plain = oddsB.body.match(/\b(\d{1,2}(?:[.,]\d+)?)\b/);
   762→        if (plain) {
   763→          const v = Number(plain[1].replace(",", "."));
   764→          if (v >= 1.01 && v <= 100) oddsVal = String(v);
   765→        }
   766→      }
   767→      if (oddsVal) {
   768→        values.odds = oddsVal;
   769→        filled.push("cote");
   770→        warnings.push("cote importée — à contrôler");
   771→      } else {
   772→        warnings.push("cote introuvable (format non reconnu) — à saisir manuellement");
   773→      }
   774→    }
   775→    const whyB = findBullet(eBullets, /^pourquoi/);
   776→    if (whyB?.body) {
   777→      synthesisBlocks.push(`Pourquoi ce marché : ${whyB.body.replace(/\s+/g, " ").trim()}`);
   778→    }
   779→  }
   780→
   781→  /* ---- F. Verdict (avec repli sur le texte intégral) ---- */
   782→  const fText = sec("F") || cleanText(raw);
   783→  if (fText) {
   784→    const vm = fText.match(/JE\s+JOUE(?:\s+SOUS\s+CONDITION)?/i);
   785→    if (vm) {
   786→      const v = stripAccents(vm[0]).toUpperCase().replace(/\s+/g, " ");
   787→      if (v.includes("SOUS CONDITION")) {
   788→        values.verdict = "JE_JOUE_SOUS_CONDITION";
   789→        const cm = fText.match(/sous condition\s*[:\-–—]?\s*(.+)/i);
   790→        values.condition = (cm ? cm[1] : "").replace(/\s+/g, " ").trim().slice(0, 2000) || undefined;
   791→        if (!values.condition) warnings.push("condition d'exécution à préciser (verdict sous condition)");
   792→      } else {
   793→        values.verdict = "JE_JOUE";
   794→      }
   795→      filled.push("verdict");
   796→    } else {
   797→      warnings.push("verdict non détecté dans la section F");
   798→    }
   799→
   800→    const fFlat = stripAccents(fText).toLowerCase().replace(/\s+/g, " ");
   801→    const conf = fFlat.match(
   802→      new RegExp(`(?:niveau\\s+de\\s+|indice\\s+de\\s+)?confiance\\s*[:\\-–—]?\\s*(${CONF_LEVEL})(?:\\s*(?:a|ou|/|->|→)\\s*(${CONF_LEVEL}))?`)
   803→    );
   804→    if (conf) {
   805→      values.confidence = levelToConfidence(conf[1]);
   806→      filled.push("niveau de confiance");
   807→      if (conf[2]) {
   808→        warnings.push(
   809→          `confiance ambiguë « ${conf[1]} à ${conf[2]} » — premier niveau appliqué, à confirmer`
   810→        );
   811→      }
   812→    } else {
   813→      warnings.push("niveau de confiance non détecté");
   814→    }
   815→
   816→    const fBullets = parseBullets(fText);
   817→    const reasonsB = findBullet(fBullets, /^raisons d[ée]terminantes/);
   818→    if (reasonsB?.body) {
   819→      synthesisBlocks.push(`Raisons déterminantes : ${reasonsB.body.replace(/\s+/g, " ").trim()}`);
   820→    }
   821→    const limitB = findBullet(fBullets, /^principale limite/);
   822→    if (limitB?.body) {
   823→      synthesisBlocks.push(`Principale limite : ${limitB.body.replace(/\s+/g, " ").trim()}`);
   824→    }
   825→  }
   826→
   827→  /* ---- G. Synthèse finale (prioritaire sur les blocs E/F) ----
   828→     Uniquement si des sections ont été détectées : en collage plat sans
   829→     sections, sec("G") renverrait le texte ENTIER — on ne l'utilise pas. */
   830→  const gText = hasSections ? sec("G") : "";
   831→  if (gText && gText.replace(/\W/g, "").length >= 60) {
   832→    const g = gText
   833→      .split("\n")
   834→      .map((l) => l.replace(/^\s*[-–—•*]\s*/, "").trim())
   835→      .filter(Boolean)
   836→      .join("\n")
   837→      .trim();
   838→    values.synthesis = capText(g, MAX_SYNTHESIS, warnings, "synthèse");
   839→    filled.push("synthèse finale (G)");
   840→  } else if (synthesisBlocks.length) {
   841→    values.synthesis = capText(synthesisBlocks.join("\n\n").trim(), MAX_SYNTHESIS, warnings, "synthèse");
   842→    filled.push("synthèse finale");
   843→  }
   844→
   845→  /* ---- Repli global par étiquettes ------------------------------------
   846→     Pour les collages sans structure reconnaissable (texte aplati, sections
   847→     numérotées non détectées, labels éclatés) : chaque champ encore vide est
   848→     recherché dans le TEXTE ENTIER par son étiquette. La valeur s'arrête au
   849→     prochain label connu, à la fin de ligne ou à un saut de section. */
   850→  if (filled.length === 0 || !values.market || !values.selection || !values.odds || !values.matchDate || !values.matchName) {
   851→    // Parcourt chaque ligne : « Label : valeur » — la valeur s'étend jusqu'au
   852→    // prochain label connu sur la MÊME ligne (collage aplati) ou fin de ligne.
   853→    const rawLines = cleanText(raw).split("\n");
   854→    const labeled: { label: string; body: string }[] = [];
   855→    for (const line of rawLines) {
   856→      const t = stripInline(line).trim();
   857→      if (!t) continue;
   858→      const lm = t.match(LABEL_LINE_RE);
   859→      if (lm) {
   860→        labeled.push({ label: normLabel(lm[1]), body: lm[2].replace(STOP_LABEL_RE, "").trim() });
   861→      } else if (labeled.length && !bulletReTop.test(t)) {
   862→        // continuation : rattachée au dernier label (texte multi-lignes)
   863→        labeled[labeled.length - 1].body += ` ${t.replace(STOP_LABEL_RE, "").trim()}`;
   864→      }
   865→    }
   866→    const findL = (re: RegExp) => labeled.find((b) => re.test(b.label))?.body;
   867→
   868→    if (!values.matchName) {
   869→      const v = findL(/^(match|rencontre|affiche|evenement)$/);
   870→      if (v) {
   871→        values.matchName = v.replace(/[.]\s*$/, "").slice(0, 160);
   872→        filled.push("affiche (repli)");
   873→      }
   874→    }
   875→    if (!values.sport || !values.competition) {
   876→      const v = findL(/^sport/) || findL(/^competition/);
   877→      if (v && !values.competition) {
   878→        const comp = extractCompetition(v);
   879→        if (comp.competition) values.competition = comp.competition.slice(0, 120);
   880→        if (comp.sportHint && !values.sport) values.sport = comp.sportHint;
   881→        if (!values.competition) values.competition = cleanCompetitionLabel(v).slice(0, 120) || undefined;
   882→        if (values.competition) filled.push("compétition (repli)");
   883→      }
   884→      if (!values.sport && values.competition) {
   885→        const f = stripAccents(values.competition).toLowerCase();
   886→        if (/\bfoot/.test(f)) values.sport = "FOOTBALL";
   887→        else if (/\bbasket/.test(f)) values.sport = "BASKETBALL";
   888→        else if (/\btennis/.test(f)) values.sport = "TENNIS";
   889→        if (values.sport) filled.push("sport (repli)");
   890→      }
   891→    }
   892→    if (!values.matchDate) {
   893→      const v = findL(/^(date( et heure| du match)?|heure|jour)$/);
   894→      if (v) {
   895→        const local = parseMatchDate(v, warnings);
   896→        if (local) {
   897→          values.matchDate = local;
   898→          filled.push("date du match (repli)");
   899→          warnings.push("date/heure importée — à contrôler");
   900→        }
   901→      }
   902→    }
   903→    if (!values.market) {
   904→      const v = findL(/^(marche|market|pari retenu)$/);
   905→      if (v) {
   906→        values.market = v.split(/\s+\(ou\b/i)[0].replace(/[;.]+$/, "").trim().slice(0, 120) || undefined;
   907→        if (values.market) filled.push("marché (repli)");
   908→      }
   909→    }
   910→    if (!values.selection) {
   911→      const v = findL(/^(selection|pronostic|pari|choix|pick)$/);
   912→      if (v) {
   913→        values.selection = v.replace(/[;.]+$/, "").trim().slice(0, 120) || undefined;
   914→        if (values.selection) filled.push("sélection (repli)");
   915→      }
   916→    }
   917→    // « Marché » contenant déjà le pari (ex. « Marché : Under 2,5 ») :
   918→    // si la sélection manque, la dériver du marché plutôt que la perdre.
   919→    if (!values.selection && values.market && values.market.length <= 40) {
   920→      values.selection = values.market;
   921→      filled.push("sélection (= marché)");
   922→      warnings.push("sélection déduite du marché — vérifiez-la");
   923→    }
   924→    if (!values.line) {
   925→      const v = findL(/^ligne$/);
   926→      const lm = v?.match(/^\s*([+\-−–—]?\s?\d+(?:[.,]\d+)?)/);
   927→      if (lm) {
   928→        values.line = lm[1].replace(/[−–—]/g, "-").replace(/\s+/g, "").slice(0, 60);
   929→        filled.push("ligne (repli)");
   930→      }
   931→    }
   932→    if (!values.odds) {
   933→      const v = findL(/^cotes?$/);
   934→      if (v) {
   935→        const dec = v.match(/\d{1,2}[.,]\d{1,3}/);
   936→        if (dec) {
   937→          values.odds = dec[0].replace(",", ".");
   938→          filled.push("cote (repli)");
   939→          warnings.push("cote importée — à contrôler");
   940→        }
   941→      }
   942→    }
   943→    if (!values.verdict) {
   944→      const vm3 = flatRawAll.match(/JE\s+JOUE(?:\s+SOUS\s+CONDITION)?/i);
   945→      if (vm3) {
   946→        const v = stripAccents(vm3[0]).toUpperCase().replace(/\s+/g, " ");
   947→        values.verdict = v.includes("SOUS CONDITION") ? "JE_JOUE_SOUS_CONDITION" : "JE_JOUE";
   948→        filled.push("verdict (repli)");
   949→      }
   950→    }
   951→    if (!values.confidence) {
   952→      const cm = flatRawAll.match(
   953→        new RegExp(`(?:niveau\\s+de\\s+|indice\\s+de\\s+)?confiance\\s*[:\\-–—]?\\s*(${CONF_LEVEL})(?:\\s*(?:a|ou|/|->|→)\\s*(${CONF_LEVEL}))?`, "i")
   954→      );
   955→      if (cm) {
   956→        values.confidence = levelToConfidence(stripAccents(cm[1]).toLowerCase());
   957→        filled.push("niveau de confiance (repli)");
   958→        if (cm[2]) warnings.push(`confiance ambiguë « ${cm[1]} à ${cm[2]} » — premier niveau appliqué`);
   959→      }
   960→    }
   961→  }
   962→
   963→  /* ---- Explicitation des champs restés vides (aucun échec silencieux) ---- */
   964→  if (filled.length >= 2) {
   965→    const criticals: [string, string | undefined][] = [
   966→      ["affiche (nom du match)", values.matchName],
   967→      ["date du match", values.matchDate],
   968→      ["marché", values.market],
   969→      ["sélection", values.selection],
   970→      ["cote", values.odds],
   971→    ];
   972→    for (const [label, v] of criticals) {
   973→      if (!v) warnings.push(`${label} non reconnu(e) — à remplir manuellement`);
   974→    }
   975→  }
   976→
   977→  /* ---- Dernier repli pour l'affiche : motif « X vs Y » ---- */
   978→  if (!values.matchName) {
   979→    const vm2 = cleanText(raw)
   980→      .slice(0, 600)
   981→      .match(/([A-Za-zÀ-ÿ0-9.''\- ]{3,48})\s+\bvs\.?\b\s+([A-Za-zÀ-ÿ0-9.''\- ]{3,48})/);
   982→    if (vm2) {
   983→      values.matchName = `${vm2[1].trim()} vs ${vm2[2].trim()}`.slice(0, 160);
   984→      filled.push("affiche (vs)");
   985→    }
   986→  }
   987→
   988→  if (filled.length === 0) {
   989→    parseErrors.push("aucun champ reconnu — vérifiez que le texte suit le format NODE.47 (sections A à G)");
   990→  }
   991→
   992→  return { values, filled, warnings, parseErrors };
   993→}
