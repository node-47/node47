   207→   207→function isSectionTitle(text: string): boolean {
   208→   208→  const t = stripAccents(text)
   209→   209→    .toLowerCase()
   210→   210→    .replace(/[^a-z& ]+/g, " ")
   211→   211→    .replace(/\s+/g, " ")
   212→   212→    .trim();
   213→   213→  return SECTION_KEYWORDS.some((k) => t.startsWith(k));
   214→   214→}

   216→   216→/** Découpe le texte en sections nommées (A à H). Accepte :
   217→   217→ *  - lettres « A. / B) / C - » (majuscules ou minuscules, avec ou sans gras)
   218→   218→ *  - numérotation quelconque « 1. 2. » « I. II. » si le titre est reconnu
   219→   219→ *  - titres nus « MATCH ET CONTEXTE » sans aucun préfixe
   220→   220→ *  Sans en-têtes reconnaissables → tout dans "ALL" (remplissage par étiquettes). */
   221→   221→function splitSections(text: string): Record<string, string> {
   222→   222→  const lines = cleanText(text).split("\n");
   223→   223→  const sections: Record<string, string[]> = { ALL: [] };
   224→   224→  let current = "ALL";
   225→   225→  const headerRe = /^\s*([A-Ha-h])\s*[.)\]:\-–—]\s*(.+?)\s*$/;
   226→   226→  const bulletRe = /^\s*[•‣▪◦*\-–—]\s+/;
   227→   227→  for (const line of lines) {
   228→   228→    const stripped = stripInline(line);
   229→   229→    const m = stripped.match(headerRe);
   230→   230→    const letters = m ? m[2].replace(/[^A-Za-zÀ-ÿ]/g, "") : "";
   231→   231→    const isHeader =
   232→   232→      !!m && letters.length > 2 && (letters === letters.toUpperCase() || isSectionTitle(m[2]));
   233→   233→    if (isHeader && m) {
   234→   234→      current = m[1].toUpperCase();
   235→   235→      if (!sections[current]) sections[current] = [];
   236→   236→      continue;
   237→   237→    }
   238→   238→    // En-tête sans lettre reconnue (numéroté ou nu) : titre court, pas de puce,
   239→   239→    // pas de « : » — la position dans l'ordre canonique tranche la section.
   240→   240→    if (!bulletRe.test(stripped) && !/:/.test(stripped) && stripped.length < 80) {
   241→   241→      const t = stripAccents(stripped)
   242→   242→        .toLowerCase()
   243→   243→        .replace(/^\s*(?:\d{1,2}|[ivxlIVXL]{1,5})\s*[.)\]:\-–—]?\s*/, "")
   244→   244→        .replace(/\s+/g, " ")
   245→   245→        .trim();
   246→   246→      const hit = ORDER_KEYWORDS.find(([re]) => re.test(t));
   247→   247→      if (hit) {
   248→   248→        current = hit[1];
   249→   249→        if (!sections[current]) sections[current] = [];
   250→   250→        continue;
   251→   251→      }
   252→   252→    }
   253→   253→    sections[current].push(stripped);
   254→   254→  }
   255→   255→  const out: Record<string, string> = {};
   256→   256→  for (const [k, v] of Object.entries(sections)) out[k] = v.join("\n").trim();
   257→   257→  return out;
   258→   258→}

   260→   260→/** Découpe une section en puces « Label : corps » (•, -, –, —, *) ; les lignes
   261→   261→    simples sont rattachées à la puce précédente (continuation / sous-listes). */
   262→   262→function parseBullets(section: string): { label: string; body: string }[] {
   263→   263→  const lines = cleanText(section).split("\n");
   264→   264→  const bullets: { label: string; body: string }[] = [];
   265→   265→  const bulletRe = /^\s*[•‣▪◦*\-–—]\s+(.*)$/;
   266→   266→  for (const line of lines) {
   267→   267→    const m = line.match(bulletRe);
   268→   268→    if (m) {
   269→   269→      const t = m[1].trim();
   270→   270→      const ci = t.indexOf(":");
   271→   271→      if (ci > 0 && ci < 80) {
   272→   272→        bullets.push({ label: t.slice(0, ci).trim(), body: t.slice(ci + 1).trim() });
   273→   273→      } else {
   274→   274→        bullets.push({ label: "", body: t });
   275→   275→      }
   276→   276→    } else if (line.trim()) {
   277→   277→      const t = line.trim();
   278→   278→      // Ligne « Label : valeur » SANS puce (collage sans mise en forme) :
   279→   279→      // traitée comme une puce étiquetée plutôt que comme du prose perdue.
   280→   280→      const lm = t.match(LABEL_LINE_RE);
   281→   281→      if (lm) {
   282→   282→        bullets.push({ label: lm[1].trim(), body: lm[2].replace(STOP_LABEL_RE, "").trim() });
   283→   283→      } else if (bullets.length) {
   284→   284→        bullets[bullets.length - 1].body += `\n${t}`;
   285→   285→      } else {
   286→   286→        bullets.push({ label: "", body: t });
   287→   287→      }
   288→   288→    }
   289→   289→  }
   290→   290→  return bullets;
   291→   291→}

   302→   302→function splitReasons(body: string): string[] {
   303→   303→  const lines = body
   304→   304→    .split("\n")
   305→   305→    .map((l) => l.replace(/^\s*[-–—•*]\s*/, "").replace(/^\d+\s*[.)]\s*/, "").trim())
   306→   306→    .filter(Boolean);
   307→   307→  if (lines.length > 1) return lines;
   308→   308→  const flat = body.replace(/\s+/g, " ").trim();
   309→   309→  if (!flat) return [];
   310→   310→  if (flat.length <= 220) return [flat];
   311→   311→  const sentences = flat
   312→   312→    .split(/(?<=[.!?])\s+/)
   313→   313→    .map((s) => s.trim())
   314→   314→    .filter((s) => s.length >= 24);
   315→   315→  return sentences.length > 1 ? sentences : [flat];
   316→   316→}

   329→   329→function extractCompetition(body: string): { competition?: string; sportHint?: string } {
   330→   330→  const flat = stripAccents(body).toLowerCase().replace(/\s+/g, " ");
   331→   331→  for (const c of COMPETITIONS) {
   332→   332→    if (c.re.test(flat)) return { competition: c.label, sportHint: c.sport };
   333→   333→  }
   334→   334→  // Repli : partie après le séparateur « / » puis « – » / « — » / « - »
   335→   335→  let tail = "";
   336→   336→  const slashParts = body.split(/\s*\/\s*/);
   337→   337→  if (slashParts.length > 1) tail = slashParts.slice(1).join(" / ");
   338→   338→  if (!tail) {
   339→   339→    const dashParts = body.split(/\s+[–—-]\s+/);
   340→   340→    if (dashParts.length > 1) tail = dashParts.slice(1).join(" — ");
   341→   341→  }
   342→   342→  if (!tail) return {};
   343→   343→  tail = cleanCompetitionLabel(tail);
   344→   344→  if (tail.length >= 3 && tail.length <= 80) return { competition: tail };
   345→   345→  return {};
   346→   346→}

   348→   348→/* ---------- Date & heure ---------- */
   349→   349→
   350→   350→function parseMatchDate(body: string, warnings: string[]): string | undefined {
   351→   351→  const flat = body.replace(/\n/g, " ");
   352→   352→  const monthKeys = Object.keys(MONTHS).sort((a, b) => b.length - a.length).join("|");
   353→   353→
   354→   354→  // Format littéraire : « 17 octobre 2026 » / « 3 novembre 2025 »
   355→   355→  let day: number | undefined;
   356→   356→  let month: number | undefined;
   357→   357→  let year: number | undefined;
   358→   358→  const dm = flat.match(new RegExp(`(\\d{1,2})(?:er)?\\s+(${monthKeys})\\s+(\\d{4})`, "i"));
   359→   359→  if (dm) {
   360→   360→    day = parseInt(dm[1], 10);
   361→   361→    month = MONTHS[dm[2].toLowerCase()];
   362→   362→    year = parseInt(dm[3], 10);
   363→   363→  } else {
   364→   364→    // Formats numériques : 17/10/2026, 17-10-2026, 17.10.2026, 2026-10-17
   365→   365→    const ymd = flat.match(/\b(\d{4})[-/](\d{1,2})[-/](\d{1,2})\b/);
   366→   366→    const dmy = ymd ? undefined : flat.match(/\b(\d{1,2})[/.‐-](\d{1,2})[/.‐-](\d{4})\b/);
   367→   367→    const parts = ymd ?? dmy;
   368→   368→    if (parts) {
   369→   369→      const a = parseInt(parts[1], 10);
   370→   370→      const b = parseInt(parts[2], 10);
   371→   371→      const c = parseInt(parts[3], 10);
   372→   372→      if (ymd) {
   373→   373→        year = a; month = b; day = c;
   374→   374→      } else {
   375→   375→        // Ambiguïté jour/mois : la valeur > 12 désigne forcément le jour
   376→   376→        day = a > 12 ? a : b > 12 ? b : a;
   377→   377→        month = day === a ? b : a;
   378→   378→        year = c;
   379→   379→      }
   380→   380→    }
   381→   381→  }
   382→   382→  if (day === undefined || month === undefined || year === undefined) return undefined;
   383→   383→  if (month < 1 || month > 12 || day < 1 || day > 31) return undefined;
   384→   384→
   385→   385→  let hour = 12;
   386→   386→  let minute = 0;
   387→   387→  const tm = flat.match(/(\d{1,2})\s*[h:]\s*(\d{2})?/i);
   388→   388→  let hasTime = false;
   389→   389→  if (tm) {
   390→   390→    hour = parseInt(tm[1], 10);
   391→   391→    minute = tm[2] ? parseInt(tm[2], 10) : 0;
   392→   392→    hasTime = true;
   393→   393→  }
   394→   394→
   395→   395→  const tzm = flat.match(
   396→   396→    /\b(CEST|CETE?|WEST|WET|EEST|EET|PDT|PST|MDT|MST|CDT|CST|EDT|EST|UTC|GMT)\b/i
   397→   397→  );
   398→   398→  let offset = 0;
   399→   399→  if (tzm) {
   400→   400→    offset = TZ_OFFSETS[tzm[1].toUpperCase()] ?? 0;
   401→   401→  } else {
   402→   402→    warnings.push("fuseau horaire non détecté — heure lue comme UTC, vérifiez la date");
   403→   403→  }
   404→   404→  if (!hasTime) {
   405→   405→    warnings.push("heure du match non détectée — 12 h placée par défaut");
   406→   406→  }
   407→   407→
   408→   408→  const utcMs = Date.UTC(year, month - 1, day, hour, minute) - offset * 3_600_000;
   409→   409→  return toDatetimeLocal(new Date(utcMs));
   410→   410→}

   412→   412→/* ---------- Statistiques clés (section C) ---------- */
   413→   413→
   414→   414→/**
   415→   415→ * Extraction prudente : seules les statistiques « propres » sont conservées
   416→   416→ * (étiquette courte et lisible devant le nombre, ou étiquette après « : »).
   417→   417→ * Les fragments de prose sont ignorés (comptés dans `skipped`) afin de ne pas
   418→   418→ * polluer le public — le texte brut reste disponible dans le scénario.
   419→   419→ */
   420→   420→function parseKeyStats(body: string): { stats: KeyStatItem[]; skipped: number } {
   421→   421→  // Protège les décimales (2,33) avant découpe sur les virgules
   422→   422→  const flat = body.replace(/\s+/g, " ").replace(/(\d),(\d)/g, "$1\u0001$2");
   423→   423→  const clauses = flat
   424→   424→    .split(/\s*[;.]\s+|\s*[;]\s*|\s*,\s+/)
   425→   425→    .map((c) => c.replace(/\u0001/g, ",").trim())
   426→   426→    .filter((c) => stripAccents(c).length >= 6);
   427→   427→
   428→   428→  const stats: KeyStatItem[] = [];
   429→   429→  let skipped = 0;
   430→   430→  for (const clause of clauses) {
   431→   431→    if (stats.length >= MAX_POINTS) break;
   432→   432→    const ci = clause.indexOf(":");
   433→   433→    const base = ci > 0 ? clause.slice(ci + 1) : clause;
   434→   434→
   435→   435→    const numRe = /([+\-−~]?\s?\d[\d\u0001\u202f ]*(?:[.,]\d+)?)/;
   436→   436→    const nm = base.match(numRe);
   437→   437→    if (!nm) continue;
   438→   438→    const after = base.slice((nm.index ?? 0) + nm[0].length).trim();
   439→   439→    const unit = after.match(/^(%|u\b|buts?|jeux?|points?|pts?\b|match(?:e)?s?|set|sets|aces?|fautes?)/i);
   440→   440→    const unitTxt = unit ? unit[0] : "";
   441→   441→    const numStr = nm[0].replace(/\u0001/g, ",").replace(/\s{2,}/g, " ").trim();
   442→   442→
   443→   443→    // Année isolée (1900-2099) sans unité → pas une statistique
   444→   444→    if (!unitTxt && /^(19|20)\d{2}$/.test(numStr)) {
   445→   445→      skipped++;
   446→   446→      continue;
   447→   447→    }
   448→   448→
   449→   449→    let label: string;
   450→   450→    if (ci > 0) {
   451→   451→      label = clause.slice(0, ci).trim();
   452→   452→      if (label.length > 64 || !/[A-Za-zÀ-ÿ]/.test(label)) {
   453→   453→        skipped++;
   454→   454→        continue;
   455→   455→      }
   456→   456→    } else {
   457→   457→      const before = base.slice(0, nm.index ?? 0).trim();
   458→   458→      if (before.length < 2 || before.length > 48 || before.includes("(") || !/[A-Za-zÀ-ÿ]/.test(before)) {
   459→   459→        skipped++;
   460→   460→        continue;
   461→   461→      }
   462→   462→      label = before;
   463→   463→    }
   464→   464→    label = label
   465→   465→      .replace(/[\s:;,–—-]+$/, "")
   466→   466→      .replace(/\s+(dans|de|du|des|la|le|les|un|une|en|sur|par|pour|avec|et|ou|au|aux)\s*$/i, "")
   467→   467→      .replace(/[\s:;,–—-]+$/, "")
   468→   468→      .trim();
   469→   469→    if (label.length < 2) {
   470→   470→      skipped++;
   471→   471→      continue;
   472→   472→    }
   473→   473→
   474→   474→    const detail = (unitTxt ? after.slice(unitTxt.length) : after).replace(/^[\s,%–—-]+/, "").trim();
   475→   475→    stats.push({
   476→   476→      label: label.slice(0, 120),
   477→   477→      value: (numStr + (unitTxt ? ` ${unitTxt}` : "")).slice(0, 60),
   478→   478→      ...(detail && detail !== label ? { detail: detail.slice(0, 300) } : {}),
   479→   479→    });
   480→   480→  }
   481→   481→  return { stats, skipped };
   482→   482→}

   719→   719→      filled.push(`points clés (${capped.length})`);
   720→   720→      if (cleaned.length > MAX_POINTS) {
   721→   721→        warnings.push(`points clés limités à ${MAX_POINTS} — ${cleaned.length - MAX_POINTS} ignorés`);
   722→   722→      }
   723→   723→    }
   724→   724→  }
   725→   725→
   726→   726→  /* ---- E. Marché retenu ---- */
   727→   727→  const eBullets = bulletsOf("E");
   728→   728→  if (eBullets.length) {
   729→   729→    const marketB = findBullet(eBullets, /^(marche|market|marche\s+retenu|pari\s+retenu)\b/);
   730→   730→    if (marketB?.body) {
   731→   731→      let mk = marketB.body.replace(/\s+/g, " ").trim();
   732→   732→      mk = mk.split(/\s+\(ou\b/i)[0].trim(); // retire le second choix éventuel
   733→   733→      mk = mk.replace(/[;.。]+\s*$/, "").trim();
   734→   734→      values.market = mk.slice(0, 120);
   735→   735→      filled.push("marché");
   736→   736→    }
   737→   737→    const selB = findBullet(eBullets, /^(selection|pronostic|pari|choix|pick|pronostic\s+retenu)\b/);
   738→   738→    if (selB?.body) {
   739→   739→      values.selection = selB.body
   740→   740→        .replace(/\s+/g, " ")
   741→   741→        .trim()
   742→   742→        .replace(/[;.。]+\s*$/, "")
   743→   743→        .slice(0, 120);
   744→   744→      filled.push("sélection");
   745→   745→    }
   746→   746→    const lineB = findBullet(eBullets, /^ligne\b/);
   747→   747→    if (lineB?.body) {
   748→   748→      const lm = lineB.body.match(/^\s*([+\-−–—]?\s?\d+(?:[.,]\d+)?)/);
   749→   749→      if (lm) {
   750→   750→        values.line = lm[1].replace(/[−–—]/g, "-").replace(/\s+/g, "").slice(0, 60);
   751→   751→        filled.push("ligne");
   752→   752→      }
   753→   753→    }
   754→   754→    const oddsB = findBullet(eBullets, /^cote/);
   755→   755→    if (oddsB?.body) {
   756→   756→      const dec = oddsB.body.match(/\d{1,2}[.,]\d{1,3}/);
   757→   757→      let oddsVal: string | undefined;
   758→   758→      if (dec) {
   759→   759→        oddsVal = dec[0].replace(",", ".");
   760→   760→      } else {
   761→   761→        const plain = oddsB.body.match(/\b(\d{1,2}(?:[.,]\d+)?)\b/);
   762→   762→        if (plain) {
   763→   763→          const v = Number(plain[1].replace(",", "."));
   764→   764→          if (v >= 1.01 && v <= 100) oddsVal = String(v);
   765→   765→        }
   766→   766→      }
   767→   767→      if (oddsVal) {
   768→   768→        values.odds = oddsVal;
   769→   769→        filled.push("cote");
   770→   770→        warnings.push("cote importée — à contrôler");
   771→   771→      } else {
   772→   772→        warnings.push("cote introuvable (format non reconnu) — à saisir manuellement");
   773→   773→      }
   774→   774→    }
   775→   775→    const whyB = findBullet(eBullets, /^pourquoi/);
   776→   776→    if (whyB?.body) {
   777→   777→      synthesisBlocks.push(`Pourquoi ce marché : ${whyB.body.replace(/\s+/g, " ").trim()}`);
   778→   778→    }
   779→   779→  }
   780→   780→
   781→   781→  /* ---- F. Verdict (avec repli sur le texte intégral) ---- */
   782→   782→  const fText = sec("F") || cleanText(raw);
   783→   783→  if (fText) {
   784→   784→    const vm = fText.match(/JE\s+JOUE(?:\s+SOUS\s+CONDITION)?/i);
   785→   785→    if (vm) {
   786→   786→      const v = stripAccents(vm[0]).toUpperCase().replace(/\s+/g, " ");
   787→   787→      if (v.includes("SOUS CONDITION")) {
   788→   788→        values.verdict = "JE_JOUE_SOUS_CONDITION";
   789→   789→        const cm = fText.match(/sous condition\s*[:\-–—]?\s*(.+)/i);
   790→   790→        values.condition = (cm ? cm[1] : "").replace(/\s+/g, " ").trim().slice(0, 2000) || undefined;
   791→   791→        if (!values.condition) warnings.push("condition d'exécution à préciser (verdict sous condition)");
   792→   792→      } else {
   793→   793→        values.verdict = "JE_JOUE";
   794→   794→      }
   795→   795→      filled.push("verdict");
   796→   796→    } else {
   797→   797→      warnings.push("verdict non détecté dans la section F");
   798→   798→    }
   799→   799→
   800→   800→    const fFlat = stripAccents(fText).toLowerCase().replace(/\s+/g, " ");
   801→   801→    const conf = fFlat.match(
   802→   802→      new RegExp(`(?:niveau\\s+de\\s+|indice\\s+de\\s+)?confiance\\s*[:\\-–—]?\\s*(${CONF_LEVEL})(?:\\s*(?:a|ou|/|->|→)\\s*(${CONF_LEVEL}))?`)
   803→   803→    );
   804→   804→    if (conf) {
   805→   805→      values.confidence = levelToConfidence(conf[1]);
   806→   806→      filled.push("niveau de confiance");
   807→   807→      if (conf[2]) {
   808→   808→        warnings.push(
   809→   809→          `confiance ambiguë « ${conf[1]} à ${conf[2]} » — premier niveau appliqué, à confirmer`
   810→   810→        );
   811→   811→      }
   812→   812→    } else {
   813→   813→      warnings.push("niveau de confiance non détecté");
   814→   814→    }
   815→   815→
   816→   816→    const fBullets = parseBullets(fText);
   817→   817→    const reasonsB = findBullet(fBullets, /^raisons d[ée]terminantes/);
   818→   818→    if (reasonsB?.body) {
   819→   819→      synthesisBlocks.push(`Raisons déterminantes : ${reasonsB.body.replace(/\s+/g, " ").trim()}`);
   820→   820→    }
   821→   821→    const limitB = findBullet(fBullets, /^principale limite/);
   822→   822→    if (limitB?.body) {
   823→   823→      synthesisBlocks.push(`Principale limite : ${limitB.body.replace(/\s+/g, " ").trim()}`);
   824→   824→    }
   825→   825→  }
   826→   826→
   827→   827→  /* ---- G. Synthèse finale (prioritaire sur les blocs E/F) ----
   828→   828→     Uniquement si des sections ont été détectées : en collage plat sans
   829→   829→     sections, sec("G") renverrait le texte ENTIER — on ne l'utilise pas. */
   830→   830→  const gText = hasSections ? sec("G") : "";
   831→   831→  if (gText && gText.replace(/\W/g, "").length >= 60) {
   832→   832→    const g = gText
   833→   833→      .split("\n")
   834→   834→      .map((l) => l.replace(/^\s*[-–—•*]\s*/, "").trim())
   835→   835→      .filter(Boolean)
   836→   836→      .join("\n")
   837→   837→      .trim();
   838→   838→    values.synthesis = capText(g, MAX_SYNTHESIS, warnings, "synthèse");
   839→   839→    filled.push("synthèse finale (G)");
   840→   840→  } else if (synthesisBlocks.length) {
   841→   841→    values.synthesis = capText(synthesisBlocks.join("\n\n").trim(), MAX_SYNTHESIS, warnings, "synthèse");
   842→   842→    filled.push("synthèse finale");
   843→   843→  }
   844→   844→
   845→   845→  /* ---- Repli global par étiquettes ------------------------------------
   846→   846→     Pour les collages sans structure reconnaissable (texte aplati, sections
   847→   847→     numérotées non détectées, labels éclatés) : chaque champ encore vide est
   848→   848→     recherché dans le TEXTE ENTIER par son étiquette. La valeur s'arrête au
   849→   849→     prochain label connu, à la fin de ligne ou à un saut de section. */
   850→   850→  if (filled.length === 0 || !values.market || !values.selection || !values.odds || !values.matchDate || !values.matchName) {
   851→   851→    // Parcourt chaque ligne : « Label : valeur » — la valeur s'étend jusqu'au
   852→   852→    // prochain label connu sur la MÊME ligne (collage aplati) ou fin de ligne.
   853→   853→    const rawLines = cleanText(raw).split("\n");
   854→   854→    const labeled: { label: string; body: string }[] = [];
   855→   855→    for (const line of rawLines) {
   856→   856→      const t = stripInline(line).trim();
   857→   857→      if (!t) continue;
   858→   858→      const lm = t.match(LABEL_LINE_RE);
   859→   859→      if (lm) {
   860→   860→        labeled.push({ label: normLabel(lm[1]), body: lm[2].replace(STOP_LABEL_RE, "").trim() });
   861→   861→      } else if (labeled.length && !bulletReTop.test(t)) {
   862→   862→        // continuation : rattachée au dernier label (texte multi-lignes)
   863→   863→        labeled[labeled.length - 1].body += ` ${t.replace(STOP_LABEL_RE, "").trim()}`;
   864→   864→      }
   865→   865→    }
   866→   866→    const findL = (re: RegExp) => labeled.find((b) => re.test(b.label))?.body;
   867→   867→
   868→   868→    if (!values.matchName) {
   869→   869→      const v = findL(/^(match|rencontre|affiche|evenement)$/);
   870→   870→      if (v) {
   871→   871→        values.matchName = v.replace(/[.]\s*$/, "").slice(0, 160);
   872→   872→        filled.push("affiche (repli)");
   873→   873→      }
   874→   874→    }
   875→   875→    if (!values.sport || !values.competition) {
   876→   876→      const v = findL(/^sport/) || findL(/^competition/);
   877→   877→      if (v && !values.competition) {
   878→   878→        const comp = extractCompetition(v);
   879→   879→        if (comp.competition) values.competition = comp.competition.slice(0, 120);
   880→   880→        if (comp.sportHint && !values.sport) values.sport = comp.sportHint;
   881→   881→        if (!values.competition) values.competition = cleanCompetitionLabel(v).slice(0, 120) || undefined;
   882→   882→        if (values.competition) filled.push("compétition (repli)");
   883→   883→      }
   884→   884→      if (!values.sport && values.competition) {
   885→   885→        const f = stripAccents(values.competition).toLowerCase();
   886→   886→        if (/\bfoot/.test(f)) values.sport = "FOOTBALL";
   887→   887→        else if (/\bbasket/.test(f)) values.sport = "BASKETBALL";
   888→   888→        else if (/\btennis/.test(f)) values.sport = "TENNIS";
   889→   889→        if (values.sport) filled.push("sport (repli)");
   890→   890→      }
   891→   891→    }
   892→   892→    if (!values.matchDate) {
   893→   893→      const v = findL(/^(date( et heure| du match)?|heure|jour)$/);
   894→   894→      if (v) {
   895→   895→        const local = parseMatchDate(v, warnings);
   896→   896→        if (local) {
   897→   897→          values.matchDate = local;
   898→   898→          filled.push("date du match (repli)");
   899→   899→          warnings.push("date/heure importée — à contrôler");
   900→   900→        }
   901→   901→      }
   902→   902→    }
   903→   903→    if (!values.market) {
   904→   904→      const v = findL(/^(marche|market|pari retenu)$/);
   905→   905→      if (v) {
   906→   906→        values.market = v.split(/\s+\(ou\b/i)[0].replace(/[;.]+$/, "").trim().slice(0, 120) || undefined;
   907→   907→        if (values.market) filled.push("marché (repli)");
   908→   908→      }
   909→   909→    }
   910→   910→    if (!values.selection) {
   911→   911→      const v = findL(/^(selection|pronostic|pari|choix|pick)$/);
   912→   912→      if (v) {
   913→   913→        values.selection = v.replace(/[;.]+$/, "").trim().slice(0, 120) || undefined;
   914→   914→        if (values.selection) filled.push("sélection (repli)");
   915→   915→      }
   916→   916→    }
   917→   917→    // « Marché » contenant déjà le pari (ex. « Marché : Under 2,5 ») :
   918→   918→    // si la sélection manque, la dériver du marché plutôt que la perdre.
   919→   919→    if (!values.selection && values.market && values.market.length <= 40) {
   920→   920→      values.selection = values.market;
   921→   921→      filled.push("sélection (= marché)");
   922→   922→      warnings.push("sélection déduite du marché — vérifiez-la");
   923→   923→    }
   924→   924→    if (!values.line) {
   925→   925→      const v = findL(/^ligne$/);
   926→   926→      const lm = v?.match(/^\s*([+\-−–—]?\s?\d+(?:[.,]\d+)?)/);
   927→   927→      if (lm) {
   928→   928→        values.line = lm[1].replace(/[−–—]/g, "-").replace(/\s+/g, "").slice(0, 60);
   929→   929→        filled.push("ligne (repli)");
   930→   930→      }
   931→   931→    }
   932→   932→    if (!values.odds) {
   933→   933→      const v = findL(/^cotes?$/);
   934→   934→      if (v) {
   935→   935→        const dec = v.match(/\d{1,2}[.,]\d{1,3}/);
   936→   936→        if (dec) {
   937→   937→          values.odds = dec[0].replace(",", ".");
   938→   938→          filled.push("cote (repli)");
   939→   939→          warnings.push("cote importée — à contrôler");
   940→   940→        }
   941→   941→      }
   942→   942→    }
   943→   943→    if (!values.verdict) {
   944→   944→      const vm3 = flatRawAll.match(/JE\s+JOUE(?:\s+SOUS\s+CONDITION)?/i);
   945→   945→      if (vm3) {
   946→   946→        const v = stripAccents(vm3[0]).toUpperCase().replace(/\s+/g, " ");
   947→   947→        values.verdict = v.includes("SOUS CONDITION") ? "JE_JOUE_SOUS_CONDITION" : "JE_JOUE";
   948→   948→        filled.push("verdict (repli)");
   949→   949→      }
   950→   950→    }
   951→   951→    if (!values.confidence) {
   952→   952→      const cm = flatRawAll.match(
   953→   953→        new RegExp(`(?:niveau\\s+de\\s+|indice\\s+de\\s+)?confiance\\s*[:\\-–—]?\\s*(${CONF_LEVEL})(?:\\s*(?:a|ou|/|->|→)\\s*(${CONF_LEVEL}))?`, "i")
   954→   954→      );
   955→   955→      if (cm) {
   956→   956→        values.confidence = levelToConfidence(stripAccents(cm[1]).toLowerCase());
   957→   957→        filled.push("niveau de confiance (repli)");
   958→   958→        if (cm[2]) warnings.push(`confiance ambiguë « ${cm[1]} à ${cm[2]} » — premier niveau appliqué`);
   959→   959→      }
   960→   960→    }
   961→   961→  }
   962→   962→
   963→   963→  /* ---- Explicitation des champs restés vides (aucun échec silencieux) ---- */
   964→   964→  if (filled.length >= 2) {
   965→   965→    const criticals: [string, string | undefined][] = [
   966→   966→      ["affiche (nom du match)", values.matchName],
   967→   967→      ["date du match", values.matchDate],
   968→   968→      ["marché", values.market],
   969→   969→      ["sélection", values.selection],
   970→   970→      ["cote", values.odds],
   971→   971→    ];
   972→   972→    for (const [label, v] of criticals) {
   973→   973→      if (!v) warnings.push(`${label} non reconnu(e) — à remplir manuellement`);
   974→   974→    }
   975→   975→  }
   976→   976→
   977→   977→  /* ---- Dernier repli pour l'affiche : motif « X vs Y » ---- */
   978→   978→  if (!values.matchName) {
   979→   979→    const vm2 = cleanText(raw)
   980→   980→      .slice(0, 600)
   981→   981→      .match(/([A-Za-zÀ-ÿ0-9.''\- ]{3,48})\s+\bvs\.?\b\s+([A-Za-zÀ-ÿ0-9.''\- ]{3,48})/);
   982→   982→    if (vm2) {
   983→   983→      values.matchName = `${vm2[1].trim()} vs ${vm2[2].trim()}`.slice(0, 160);
   984→   984→      filled.push("affiche (vs)");
   985→   985→    }
   986→   986→  }
   987→   987→
   988→   988→  if (filled.length === 0) {
   989→   989→    parseErrors.push("aucun champ reconnu — vérifiez que le texte suit le format NODE.47 (sections A à G)");
   990→   990→  }
   991→   991→
   992→   992→  return { values, filled, warnings, parseErrors };
   993→   993→}
