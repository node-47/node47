    77→    77→/** Date (Date|ISO) -> "YYYY-MM-DDTHH:mm" pour input datetime-local */
    78→    78→export function toDatetimeLocal(d: Date | string): string {
    79→    79→  const date = typeof d === "string" ? new Date(d) : d;
    80→    80→  const pad = (n: number) => String(n).padStart(2, "0");
    81→    81→  return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}T${pad(date.getHours())}:${pad(date.getMinutes())}`;
    82→    82→}

   113→   113→export function PickForm({ initial, mode }: { initial?: PickFormInitial; mode: "create" | "edit" }) {
   114→   114→  const router = useRouter();
   115→   115→  const { toast } = useToast();
   116→   116→  const [form, setForm] = useState<PickFormInitial>(initial ?? EmptyPickForm());
   117→   117→  const [loading, setLoading] = useState(false);
   118→   118→  const [error, setError] = useState<string | null>(null);
   119→   119→  const [importOpen, setImportOpen] = useState(mode === "create");
   120→   120→  const [importText, setImportText] = useState("");

   122→   122→  const fileInputRef = useRef<HTMLInputElement>(null);
   123→   123→
   124→   124→  function set<K extends keyof PickFormInitial>(key: K, value: PickFormInitial[K]) {
   125→   125→    setForm((f) => ({ ...f, [key]: value }));
   126→   126→    setImportWarn((w) => (w[key] ? { ...w, [key]: false } : w));
   127→   127→  }
   128→   128→
   129→   129→  /* --- Import d'une analyse collée --- */
   130→   130→  function applyImport() {
   131→   131→    const report = parseAnalyzeText(importText);
   132→   132→    if (report.parseErrors.length && report.filled.length === 0) {
   133→   133→      setError(`Import impossible — ${report.parseErrors[0]}`);
   134→   134→      return;
   135→   135→    }
   136→   136→    // Les clés absentes du rapport ne doivent pas écraser des valeurs déjà
   137→   137→    // saisies : on retire les éventuelles valeurs undefined avant fusion.
   138→   138→    const cleanValues = Object.fromEntries(
   139→   139→      Object.entries(report.values).filter(([, v]) => v !== undefined)
   140→   140→    ) as Partial<PickFormInitial>;
   141→   141→    setForm((f) => ({ ...f, ...cleanValues }));
   142→   142→    setImportWarn({
   143→   143→      ...(report.values.matchDate ? { matchDate: true } : {}),
   144→   144→      ...(report.values.odds ? { odds: true } : {}),
   145→   145→    });
   146→   146→    const n = report.filled.length;
   147→   147→    const filledList = report.filled.slice(0, 10).join(" · ");
   148→   148→    toast({
   149→   149→      title: `Import analysé — ${n} champ${n > 1 ? "s" : ""} rempli${n > 1 ? "s" : ""}`,
   150→   150→      description: [
   151→   151→        filledList ? `Rempli : ${filledList}.` : "",
   152→   152→        report.warnings.length
   153→   153→          ? `À vérifier : ${report.warnings.slice(0, 6).join(" · ")}`
   154→   154→          : "Tous les champs restent modifiables avant enregistrement.",
   155→   155→      ]
   156→   156→        .filter(Boolean)
   157→   157→        .join(" "),
   158→   158→    });
   159→   159→    setImportText("");
   160→   160→    setImportOpen(false);
   161→   161→  }
   162→   162→
   163→   163→  /* --- Chargement d'un fichier d'analyse (.txt / .md) --- */
   164→   164→  function onFileChosen(e: React.ChangeEvent<HTMLInputElement>) {
   165→   165→    const file = e.target.files?.[0];
   166→   166→    if (!file) return;
   167→   167→    if (file.size > 200_000) {
   168→   168→      setError("Fichier trop volumineux (200 Ko maximum).");
   169→   169→      e.target.value = "";
   170→   170→      return;
   171→   171→    }
   172→   172→    const reader = new FileReader();
   173→   173→    reader.onload = () => {
   174→   174→      setImportText(String(reader.result ?? ""));
   175→   175→      setError(null);
   176→   176→      toast({
   177→   177→        title: `Fichier « ${file.name} » chargé`,
   178→   178→        description: "Cliquez sur « Analyser et remplir » pour pré-remplir le formulaire.",
   179→   179→      });
   180→   180→    };
   181→   181→    reader.onerror = () => setError("Lecture du fichier impossible.");
   182→   182→    reader.readAsText(file, "utf-8");
   183→   183→    e.target.value = "";
   184→   184→  }
   185→   185→
   186→   186→  /* --- Stats clés dynamiques --- */
   187→   187→  function updateStat(index: number, patch: Partial<KeyStatItem>) {
   188→   188→    setForm((f) => ({
   189→   189→      ...f,
   190→   190→      keyStats: f.keyStats.map((s, i) => (i === index ? { ...s, ...patch } : s)),
   191→   191→    }));
   192→   192→  }
   193→   193→  function addStat() {
   194→   194→    setForm((f) => (f.keyStats.length >= 12 ? f : { ...f, keyStats: [...f.keyStats, { label: "", value: "" }] }));
   195→   195→  }
   196→   196→  function removeStat(index: number) {
   197→   197→    setForm((f) => ({ ...f, keyStats: f.keyStats.filter((_, i) => i !== index) }));
   198→   198→  }
   199→   199→
   200→   200→  async function handleSubmit(e: React.FormEvent) {

   203→   203→
   204→   204→    /* Validation légère côté client */
   205→   205→    if (!form.competition.trim()) return setError("La compétition est requise.");

   209→   209→    const oddsNum = form.odds.trim() ? Number(form.odds.replace(",", ".")) : null;
   210→   210→    if (oddsNum !== null && (!Number.isFinite(oddsNum) || oddsNum < 1)) {
   211→   211→      return setError("La cote doit être un nombre supérieur ou égal à 1.");
   212→   212→    }
   213→   213→
   214→   214→    const keyPoints = form.keyPoints
   215→   215→      .split("\n")
   216→   216→      .map((l) => l.trim())
   217→   217→      .filter(Boolean);
   218→   218→
   219→   219→    const keyStats = form.keyStats
   220→   220→      .filter((s) => s.label.trim() && s.value.trim())
   221→   221→      .map((s) => ({
   222→   222→        label: s.label.trim(),
   223→   223→        value: s.value.trim(),
   224→   224→        ...(s.detail?.trim() ? { detail: s.detail.trim() } : {}),
   225→   225→      }));
   226→   226→
   227→   227→    const payload = {
   228→   228→      sport: form.sport,
   229→   229→      competition: form.competition.trim(),
   230→   230→      matchName: form.matchName.trim(),
   231→   231→      matchDate: new Date(form.matchDate).toISOString(),
   232→   232→      scenarioSummary: form.scenarioSummary.trim(),
   233→   233→      keyPoints,
   234→   234→      keyStats,
   235→   235→      alternativeScenario: form.alternativeScenario.trim() || null,
   236→   236→      market: form.market.trim(),
   237→   237→      selection: form.selection.trim(),
   238→   238→      line: form.line.trim() || null,
   239→   239→      odds: oddsNum,
   240→   240→      verdict: form.verdict,
   241→   241→      confidence: form.confidence,
   242→   242→      condition: form.condition.trim() || null,
   243→   243→      synthesis: form.synthesis.trim(),
   244→   244→      status: form.status,
   245→   245→      result: form.result,
   246→   246→      resultNote: form.resultNote.trim() || null,
   247→   247→    };
   248→   248→
   249→   249→    setLoading(true);
   250→   250→    try {
   251→   251→      const res =
   252→   252→        mode === "create"
   253→   253→          ? await fetch("/api/admin/picks", {
   254→   254→              method: "POST",
   255→   255→              headers: { "Content-Type": "application/json" },
   256→   256→              body: JSON.stringify(payload),
   257→   257→            })
   258→   258→          : await fetch(`/api/admin/picks/${initial?.id}`, {
   259→   259→              method: "PATCH",
   260→   260→              headers: { "Content-Type": "application/json" },
   261→   261→              body: JSON.stringify(payload),
   262→   262→            });
   263→   263→
   264→   264→      const data = await res.json().catch(() => null);
   265→   265→      if (!res.ok) {
   266→   266→        setError(data?.error ?? "Enregistrement impossible.");
   267→   267→        return;
   268→   268→      }
   269→   269→
   270→   270→      toast({
   271→   271→        title:
   272→   272→          mode === "create"
   273→   273→            ? payload.status === "PUBLISHED"
   274→   274→              ? "Analyse créée et publiée"
   275→   275→              : "Analyse créée en brouillon"
   276→   276→            : "Analyse mise à jour",
   277→   277→      });
   278→   278→      router.push("/admin");
   279→   279→      router.refresh();
   280→   280→    } catch {
   281→   281→      setError("Erreur réseau. Réessayez.");
   282→   282→    } finally {
   283→   283→      setLoading(false);
   284→   284→    }
   285→   285→  }
   286→   286→
   287→   287→  return (
   288→   288→    <form onSubmit={handleSubmit} className="space-y-6">
   289→   289→      {/* ============ 0. Import depuis un texte ============ */}
   290→   290→      <section className="rounded-xl border border-dashed border-primary/30 bg-primary/5 p-5 sm:p-6">
   291→   291→        <button
   292→   292→          type="button"
   293→   293→          onClick={() => setImportOpen((v) => !v)}
   294→   294→          className="flex w-full items-center justify-between gap-3 text-left"
   295→   295→          aria-expanded={importOpen}
   296→   296→        >
   297→   297→          <span className="flex items-center gap-2.5">
   298→   298→            <ClipboardPaste className="h-4 w-4 text-primary" aria-hidden="true" />
   299→   299→            <span className="font-display text-sm font-semibold uppercase tracking-[0.14em] text-primary">
   300→   300→              Importer depuis un texte
   301→   301→            </span>
   302→   302→          </span>
   303→   303→          <span className="text-xs text-muted-foreground">
   304→   304→            {importOpen ? "Masquer" : "Coller ou charger une analyse complète (A à G)"}
   305→   305→          </span>
   306→   306→        </button>
   307→   307→
   308→   308→        {importOpen && (
   309→   309→          <div className="mt-4 space-y-3">
   310→   310→            <Textarea
   311→   311→              value={importText}
   312→   312→              onChange={(e) => setImportText(e.target.value)}
   313→   313→              rows={10}
   314→   314→              placeholder={"Collez ou chargez ici l'analyse (sections A à G, mise en forme libre : gras, tirets, puces…).\n\nEx. : NODE.47 — Analyse : FC Porto vs PSV Eindhoven\nA. MATCH ET CONTEXTE\n- Match : …\n- Sport / compétition : Football / Ligue des Champions"}
   315→   315→              className="font-mono text-xs leading-relaxed"
   316→   316→            />
   317→   317→            <div className="flex flex-wrap items-center gap-2">
   318→   318→              <Button
   319→   319→                type="button"
   320→   320→                onClick={applyImport}
   321→   321→                disabled={!importText.trim()}
   322→   322→                className="h-9 gap-2 bg-primary px-4 text-primary-foreground hover:bg-primary/90"
   323→   323→              >
   324→   324→                <Wand2 className="h-4 w-4" aria-hidden="true" />
   325→   325→                Analyser et remplir

   331→   331→                className="h-9 gap-2"
   332→   332→              >
   333→   333→                <FileUp className="h-4 w-4" aria-hidden="true" />
   334→   334→                Charger un fichier
   335→   335→              </Button>
   336→   336→              <Button
   337→   337→                type="button"
   338→   338→                variant="ghost"
   339→   339→                onClick={() => setImportText("")}
   340→   340→                disabled={!importText}
   341→   341→                className="h-9"
   342→   342→              >
   343→   343→                Effacer
   344→   344→              </Button>
   345→   345→              <input
   346→   346→                ref={fileInputRef}
   347→   347→                type="file"
   348→   348→                accept=".txt,.md,.markdown,text/plain,text/markdown"
   349→   349→                onChange={onFileChosen}
   350→   350→                className="hidden"

   363→   363→
   364→   364→      {/* ============ 1. Match ============ */}

   367→   367→          01 — Rencontre

   373→   373→                <SelectValue placeholder="Choisir un sport" />

   382→   382→            </Select>
   383→   383→          </Field>
   384→   384→
   385→   385→          <Field label="Compétition" hint="Ex. : Premier League, NBA, ATP Masters 1000">

   424→   424→        <div className="mt-5 space-y-5">
   425→   425→          <Field label="Scénario sportif (résumé)">
