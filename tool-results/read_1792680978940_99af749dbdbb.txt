   908→      } else if (/^facteurs/.test(lab)) {
   909→        const items = b.body
   910→          .split(/\s*;\s*|\n/)
   911→          .map((l) => l.replace(/^\s*[-–—•*]\s*/, "").replace(/^\d+\s*[.)]\s*/, "").trim())
   912→          .filter((l) => l.length >= 6);
   913→        keyPointItems.push(...items);
   914→      } else if (hasSections && !b.label && b.body && b.body.replace(/\W/g, "").length >= 12) {
   915→        // Sous-puces imbriqués (ex. liste de raisons sans étiquette) —
   916→        // uniquement en mode sections : en collage plat, verser toutes les
   917→        // lignes de prose dans les points clés « ramassait tout ».
   918→        const items = b.body
   919→          .split("\n")
   920→          .map((l) => l.replace(/^\s*[-–—•*]\s*/, "").replace(/^\d+\s*[.)]\s*/, "").trim())
   921→          .filter((l) => l.length >= 6);
   922→        keyPointItems.push(...items);
   923→        bSubReasons.push(...items.slice(0, 2));
   924→      } else if (/^(equipe|joueur)\s+superieur/.test(lab) && b.body) {
   925→        supLine = `${b.label} : ${b.body.replace(/\s+/g, " ").trim().replace(/[.]\s*$/, "")}`;
   926→      } else if (/^(scenario\s+attendu|attendu)$/.test(lab) && b.body) {
   927→        attenduLine = `Scénario attendu : ${firstSentences(b.body, 320, 2)}`;
   928→      } else if (b.label && b.body && b.body.replace(/\W/g, "").length >= 12) {
   929→        // Autre puce étiquetée : une seule phrase au résumé (jamais tout).
   930→        extraBLines.push(`${b.label} : ${firstSentences(b.body, 160, 1)}`);
   931→      }
   932→      // Les puces sans étiquette en collage plat ne vont nulle part ici :
   933→      // la prose narrative est gérée par l'assemblage du scénario.
   934→    }
   935→  }
   936→
   937→  /* ---- C. Convergence des preuves ----
   938→     TOUTES les puces de C passent au détecteur de statistiques (pas
   939→     seulement « Statistiques : ») : bilans NV-ND-ND, scores, rapports,
   940→     %, volumes avec unité… Les puces marché/cotes sont écartées (leurs
   941→     valeurs alimentent les champs E). Une puce SANS statistique lisible
   942→     devient un POINT CLÉ de décision (blessures, modèles, tactique…)
   943→     au lieu d'être perdue. Un paragraphe libre de C devient la phrase
   944→     « Convergence : » du résumé. */
   945→  const cBullets = bulletsOf("C");
   946→  const statAcc: KeyStatItem[] = [];
   947→  const seenStats = new Set<string>();
   948→  let cLead = "";
   949→  let statBulletSkipped = 0;
   950→  if (hasSections) {
   951→    // Paragraphe de convergence NON à puces (synthèse libre de la section C)
   952→    const leadParts = sec("C")
   953→      .split("\n")
   954→      .map((l) => l.trim())
   955→      .filter((l) => l && !/^\s*[•‣▪◦*\-–—]/.test(l) && !LABEL_LINE_RE.test(l) && l.length > 40);
   956→    if (leadParts.length) cLead = leadParts.join(" ").replace(/\s+/g, " ").trim();
   957→
   958→    for (const b of cBullets) {
   959→      const lab = normLabel(b.label);
   960→      const bodyFlat = b.body.replace(/\s+/g, " ").trim();
   961→      const hay = stripAccents(`${b.label} ${bodyFlat}`).toLowerCase();
   962→      const isStatBullet = /^statistiques?/.test(lab);
   963→      if (MARKETISH_RE.test(hay)) continue; // marché/cotes → champs E
   964→      if (bodyFlat) {
   965→        const { stats, skipped } = parseKeyStats(bodyFlat);
   966→        if (isStatBullet) statBulletSkipped = skipped;
   967→        let added = 0;
   968→        for (const s of stats) {
   969→          const key = `${stripAccents(s.label).toLowerCase()}|${s.value}`;
   970→          if (seenStats.has(key)) continue;
   971→          seenStats.add(key);
   972→          if (statAcc.length < MAX_POINTS) {
   973→            statAcc.push(s);
   974→            added++;
   975→          }
   976→        }
   977→        if (added > 0) continue; // la puce vit dans la grille de stats
   978→      }
   979→      // Pas de statistique lisible → point clé de décision (une idée/puce)
   980→      if (bodyFlat.replace(/\W/g, "").length >= 30) {
   981→        const head = b.label ? `${b.label} : ` : "";
   982→        keyPointItems.push(`${head}${firstSentences(bodyFlat, 280, 2)}`.trim());
   983→      }
   984→    }
   985→    if (statAcc.length) {
   986→      values.keyStats = statAcc;
   987→      filled.push(`statistiques clés (${statAcc.length})`);
   988→      if (statBulletSkipped > 0) {
   989→        warnings.push(
   990→          `statistiques partielles (${statBulletSkipped} fragment(s) non décodé(s)) — vérifiez la section C`
   991→        );
   992→      }
   993→    } else if (statBulletSkipped > 0) {
   994→      warnings.push("statistiques non décodées — section C à saisir à la main");
   995→    }
   996→  }
   997→
   998→  /* ---- D. Comparaison des marchés ----
   999→     Détaillée dans les champs et points clés ; volontairement absente du
  1000→     résumé (sinon le « résumé » retrouve la longueur du texte collé). */
  1001→
  1002→  /* ---- Assemblage du scénario principal (un VRAI résumé) ----
  1003→     Sections reconnues : contexte + équipe supérieure + scénario attendu +
  1004→     premières raisons + convergence — condensé, coupé en fin de phrase.
  1005→     La comparaison des marchés (D) et le détail des puces de C restent
  1006→     hors résumé : ils vivent déjà dans les champs, les stats et les
  1007→     points clés. Sans sections (collage plat) : narration = tout SAUF
  1008→     les lignes « Label : valeur » de champs (déjà versées dans le
  1009→     formulaire) et la ligne d'affiche (déjà dans matchName), condensée
  1010→     si elle déborde. */
  1011→  const vsLineRe = /^[A-Za-zÀ-ÿ0-9.'\'\- ]{3,48}\s+vs\.?\s+[A-Za-zÀ-ÿ0-9.'\'\- ]{3,48}\s*$/;
  1012→  if (!hasSections) {
  1013→    const fieldLabelsNoNarr = FLAT_LABELS.filter((l) => l !== "equipe superieure");
  1014→    const fieldLineRe = new RegExp(
  1015→      `^(?:[-–—•*]\\s*)?(?:${fieldLabelsNoNarr.map((l) => accentPat(l).replace(/ /g, "\\s+")).join("|")})\\s*[:\\-–—]`,
  1016→      "i"
  1017→    );
  1018→    const narrative = cleanText(raw)
  1019→      .split("\n")
  1020→      .map((l) => stripInline(l).trim())
  1021→      .filter(Boolean)
  1022→      .filter((l) => !fieldLineRe.test(l))
  1023→      .filter((l) => !vsLineRe.test(l))
  1024→      // Titres d'analyse et lignes purement date/heure : ni scénario, ni prose.
  1025→      .filter((l) => !/^(?:analyse|node\.?\s*47)\b/i.test(l))
  1026→      .filter((l) => !(l.length <= 60 && DATE_LINE_RE.test(l)));
  1027→    if (narrative.length) {
  1028→      const full = narrative.join("\n");
  1029→      values.scenarioSummary = truncateAtSentence(full, SUMMARY_TARGET);
  1030→      if (values.scenarioSummary.length < full.length) {
  1031→        warnings.push("scénario condensé à l'essentiel — le texte intégral n'est pas conservé");
  1032→      }
  1033→      filled.push("scénario sportif");
  1034→
  1035→      /* Points clés de décision en collage plat : les phrases qui portent
  1036→         une justification (forme, absences, convergence, styles…). */
  1037→      const kwRe =
  1038→        /(car\b|parce\b|gr[âa]ce\b|raison|facteur|forme|absence|converge|d[ée]cisif|cl[ée]\b|avantage|manque|solide|faible|domine|sup[ée]rieur|ferm[ée]|ouvert|offensif|d[ée]fensif|buts?\b|xg\b|statistique)/i;
  1039→      const sentences = splitSentences(narrative.join(" "));
  1040→      let pts = sentences.filter(
  1041→        (s) => s.length >= 30 && s.length <= 300 && kwRe.test(stripAccents(s))
  1042→      );
  1043→      if (!pts.length) pts = sentences.filter((s) => s.length >= 30).slice(0, 3);
  1044→      keyPointItems.push(...dedupeItems(pts).slice(0, 6));
  1045→
  1046→      /* Statistiques en collage plat : détectées sur la narration (les
  1047→         lignes de champs — cote, marché… — sont exclues d'office). */
  1048→      if (!values.keyStats?.length) {
  1049→        const statSource = narrative
  1050→          .filter((l) => !MARKETISH_RE.test(stripAccents(l)))
  1051→          .join(" ");
  1052→        const { stats } = parseKeyStats(statSource);
  1053→        if (stats.length) {
  1054→          values.keyStats = stats.slice(0, 8);
  1055→          filled.push(`statistiques clés (${values.keyStats.length})`);
  1056→          warnings.push("statistiques déduites du texte — vérifiez-les");
  1057→        }
  1058→      }
  1059→    }
  1060→  } else {
  1061→    const scenChunks: string[] = [];
  1062→    if (contextLine) scenChunks.push(contextLine);
  1063→    if (supLine) scenChunks.push(supLine);
  1064→    if (attenduLine) scenChunks.push(attenduLine);
  1065→    if (raisonsLine) scenChunks.push(`Raisons : ${raisonsLine}`);
  1066→    for (const sub of bSubReasons) scenChunks.push(sub);
  1067→    for (const extra of extraBLines) scenChunks.push(extra);
  1068→    if (cLead) scenChunks.push(`Convergence : ${firstSentences(cLead, 260, 1)}`);
  1069→    const summary = scenChunks.join("\n").trim();
  1070→    if (summary) {
  1071→      values.scenarioSummary = truncateAtSentence(summary, SUMMARY_TARGET);
  1072→      if (values.scenarioSummary.length < summary.length) {
  1073→        warnings.push("scénario condensé à l'essentiel — le texte intégral n'est pas conservé");
  1074→      }
  1075→      filled.push("scénario sportif");
  1076→    }
  1077→  }
  1078→
  1079→  /* ---- E. Marché retenu ---- */
  1080→  const eBullets = bulletsOf("E");
  1081→  if (eBullets.length) {
  1082→    const marketB = findBullet(eBullets, /^(marche|market|marche\s+retenu|pari\s+retenu)\b/);
  1083→    if (marketB?.body) {
  1084→      let mk = marketB.body.replace(/\s+/g, " ").trim();
  1085→      mk = mk.split(/\s+\(ou\b/i)[0].trim(); // retire le second choix éventuel
  1086→      mk = mk.replace(/[;.。]+\s*$/, "").trim();
  1087→      values.market = mk.slice(0, 120);
  1088→      filled.push("marché");
  1089→    }
  1090→    const selB = findBullet(eBullets, /^(selection|pronostic|pari|choix|pick|pronostic\s+retenu)\b/);
  1091→    if (selB?.body) {
  1092→      values.selection = selB.body
  1093→        .replace(/\s+/g, " ")
  1094→        .trim()
  1095→        .replace(/[;.。]+\s*$/, "")
  1096→        .slice(0, 120);
  1097→      filled.push("sélection");
  1098→    }
  1099→    const lineB = findBullet(eBullets, /^ligne\b/);
  1100→    if (lineB?.body) {
  1101→      // Nombre signé en tête (« -0,5 ») ou, à défaut, premier nombre isolé
  1102→      // dans la valeur (« Under 2,5 (ou Under 2,25…) » → « 2,5 »).
  1103→      const lm = lineB.body.match(/^\s*([+\-−–—]?\s?\d+(?:[.,]\d+)?)/);
  1104→      if (lm) {
  1105→        values.line = lm[1].replace(/[−–—]/g, "-").replace(/\s+/g, "").slice(0, 60);
  1106→        filled.push("ligne");
  1107→      } else {
  1108→        const am = lineB.body.match(/([+\-−–]?\s?\d+(?:[.,]\d+)?)/);
  1109→        if (am) {
  1110→          values.line = am[1].replace(/[−–—]/g, "-").replace(/\s+/g, "").slice(0, 60);
  1111→          filled.push("ligne");
  1112→        }
  1113→      }
  1114→    }
  1115→    const oddsB = findBullet(eBullets, /^cote/);
  1116→    if (oddsB?.body) {
  1117→      const dec = oddsB.body.match(/\d{1,2}[.,]\d{1,3}/);
  1118→      let oddsVal: string | undefined;
  1119→      if (dec) {
  1120→        oddsVal = dec[0].replace(",", ".");
  1121→      } else {
  1122→        const plain = oddsB.body.match(/\b(\d{1,2}(?:[.,]\d+)?)\b/);
  1123→        if (plain) {
  1124→          const v = Number(plain[1].replace(",", "."));
  1125→          if (v >= 1.01 && v <= 100) oddsVal = String(v);
  1126→        }
  1127→      }
  1128→      if (oddsVal) {
  1129→        values.odds = oddsVal;
  1130→        filled.push("cote");
  1131→        warnings.push("cote importée — à contrôler");
  1132→      } else {
  1133→        warnings.push("cote introuvable (format non reconnu) — à saisir manuellement");
  1134→      }
  1135→    }
  1136→    const whyB = findBullet(eBullets, /^pourquoi/);
  1137→    if (whyB?.body) {
  1138→      synthesisBlocks.push(`Pourquoi ce marché : ${whyB.body.replace(/\s+/g, " ").trim()}`);
  1139→    }
  1140→  }
  1141→
  1142→  /* ---- G détecté tôt : la synthèse vient-elle de la section G ? ----
  1143→     Si oui, les raisons déterminantes de F ne seraient visibles nulle part
  1144→     sur la fiche → elles deviennent des POINTS CLÉS de décision. */
  1145→  const gText = hasSections ? sec("G") : "";
  1146→  const gUsed = !!(gText && gText.replace(/\W/g, "").length >= 60);
  1147→
  1148→  /* ---- F. Verdict (avec repli sur le texte intégral) ---- */
  1149→  const fText = sec("F") || cleanText(raw);
  1150→  if (fText) {
  1151→    const vm = fText.match(/JE\s+JOUE(?:\s+SOUS\s+CONDITION)?/i);
  1152→    if (vm) {
  1153→      const v = stripAccents(vm[0]).toUpperCase().replace(/\s+/g, " ");
  1154→      if (v.includes("SOUS CONDITION")) {
  1155→        values.verdict = "JE_JOUE_SOUS_CONDITION";
  1156→        const cm = fText.match(/sous condition\s*[:\-–—]?\s*(.+)/i);
  1157→        values.condition = (cm ? cm[1] : "").replace(/\s+/g, " ").trim().slice(0, 2000) || undefined;
  1158→        if (!values.condition) warnings.push("condition d'exécution à préciser (verdict sous condition)");
  1159→      } else {
  1160→        values.verdict = "JE_JOUE";
  1161→      }
  1162→      filled.push("verdict");
  1163→    } else {
  1164→      warnings.push("verdict non détecté dans la section F");
  1165→    }
  1166→
  1167→    const fFlat = stripAccents(fText).toLowerCase().replace(/\s+/g, " ");
  1168→    const conf = fFlat.match(
  1169→      new RegExp(`(?:niveau\\s+de\\s+|indice\\s+de\\s+)?confiance\\s*[:\\-–—]?\\s*(${CONF_LEVEL})(?:\\s*(?:a|ou|/|->|→)\\s*(${CONF_LEVEL}))?`)
  1170→    );
  1171→    if (conf) {
  1172→      values.confidence = levelToConfidence(conf[1]);
  1173→      filled.push("niveau de confiance");
  1174→      if (conf[2]) {
  1175→        warnings.push(
  1176→          `confiance ambiguë « ${conf[1]} à ${conf[2]} » — premier niveau appliqué, à confirmer`
  1177→        );
  1178→      }
  1179→    } else {
  1180→      warnings.push("niveau de confiance non détecté");
  1181→    }
  1182→
  1183→    const fBullets = parseBullets(fText);
  1184→    const reasonsB = findBullet(fBullets, /^raisons d[ée]terminantes/);
  1185→    if (reasonsB?.body) {
  1186→      synthesisBlocks.push(`Raisons déterminantes : ${reasonsB.body.replace(/\s+/g, " ").trim()}`);
  1187→      if (gUsed) keyPointItems.push(...splitReasons(reasonsB.body));
  1188→    }
  1189→    const limitB = findBullet(fBullets, /^principale limite/);
  1190→    if (limitB?.body) {
  1191→      synthesisBlocks.push(`Principale limite : ${limitB.body.replace(/\s+/g, " ").trim()}`);
  1192→    }
  1193→  }
  1194→
  1195→  /* ---- G. Synthèse finale (prioritaire sur les blocs E/F) ----
  1196→     Uniquement si des sections ont été détectées : en collage plat sans
  1197→     sections, sec("G") renverrait le texte ENTIER — on ne l'utilise pas. */
  1198→  if (gText && gText.replace(/\W/g, "").length >= 60) {
  1199→    const g = gText
  1200→      .split("\n")
  1201→      .map((l) => l.replace(/^\s*[-–—•*]\s*/, "").trim())
  1202→      .filter(Boolean)
  1203→      .join("\n")
  1204→      .trim();
  1205→    values.synthesis = capText(g, MAX_SYNTHESIS, warnings, "synthèse");
  1206→    filled.push("synthèse finale (G)");
  1207→  } else if (synthesisBlocks.length) {
  1208→    values.synthesis = capText(synthesisBlocks.join("\n\n").trim(), MAX_SYNTHESIS, warnings, "synthèse");
  1209→    filled.push("synthèse finale");
  1210→  }
  1211→
  1212→  /* ---- Points clés de décision (B + C + F + collage plat, dédoublonnés) ---- */
  1213→  if (keyPointItems.length) {
  1214→    const cleaned = dedupeItems(keyPointItems).filter((k) => !vsLineRe.test(k.trim()));
  1215→    if (cleaned.length) {
  1216→      const capped = cleaned.slice(0, MAX_POINTS).map((k) => k.slice(0, 480));
  1217→      values.keyPoints = capped.join("\n");
  1218→      filled.push(`points clés (${capped.length})`);
  1219→      if (cleaned.length > MAX_POINTS) {
  1220→        warnings.push(`points clés limités à ${MAX_POINTS} — ${cleaned.length - MAX_POINTS} ignorés`);
  1221→      }
  1222→    }
  1223→  }
  1224→
  1225→  /* ---- Repli global par étiquettes ------------------------------------
  1226→     Pour les collages sans structure reconnaissable (texte aplati, sections
  1227→     numérotées non détectées, labels éclatés) : chaque champ encore vide est
  1228→     recherché dans le TEXTE ENTIER par son étiquette. La valeur s'arrête au
  1229→     prochain label connu, à la fin de ligne ou à un saut de section. */
  1230→  if (
  1231→    filled.length === 0 || !values.market || !values.selection || !values.odds ||
  1232→    !values.matchDate || !values.matchName || !values.sport || !values.competition
  1233→  ) {
  1234→    // Parcourt chaque ligne : « Label : valeur » — la valeur s'étend jusqu'au
  1235→    // prochain label connu sur la MÊME ligne (collage aplati) ou fin de ligne.
  1236→    const rawLines = cleanText(raw).split("\n");
  1237→    const labeled: { label: string; body: string }[] = [];
  1238→    for (const line of rawLines) {
  1239→      const t = stripInline(line).trim();
  1240→      if (!t || DECOR_LINE_RE.test(t)) continue;
  1241→      const lm = t.match(LABEL_LINE_RE);
  1242→      if (lm) {
  1243→        labeled.push({ label: normLabel(lm[1]), body: lm[2].replace(STOP_LABEL_RE, "").trim() });
  1244→      } else if (
  1245→        labeled.length && !bulletReTop.test(t) &&
  1246→        !isShortLabel(labeled[labeled.length - 1].label)
  1247→      ) {
  1248→        // continuation : rattachée au dernier label (texte multi-lignes) —
  1249→        // JAMAIS pour les champs discrets (marché, sélection, cote…).
  1250→        labeled[labeled.length - 1].body += ` ${t.replace(STOP_LABEL_RE, "").trim()}`;
  1251→      }
  1252→    }
  1253→    const findL = (re: RegExp) => labeled.find((b) => re.test(b.label))?.body;
  1254→
  1255→    if (!values.matchName) {
  1256→      const v = findL(/^(match|rencontre|affiche|evenement)$/);
  1257→      if (v) {
  1258→        values.matchName = v.replace(/[.]\s*$/, "").slice(0, 160);
  1259→        filled.push("affiche (repli)");
  1260→      }
  1261→    }
  1262→    if (!values.sport || !values.competition) {
  1263→      const v = findL(/^sport/) || findL(/^competition/);
  1264→      if (v && !values.competition) {
  1265→        const comp = extractCompetition(v);
  1266→        if (comp.competition) values.competition = comp.competition.slice(0, 120);
  1267→        if (comp.sportHint && !values.sport) values.sport = comp.sportHint;
  1268→        if (!values.competition) values.competition = cleanCompetitionLabel(v).slice(0, 120) || undefined;
  1269→        if (values.competition) filled.push("compétition (repli)");
  1270→      }
  1271→      if (!values.sport && values.competition) {
  1272→        const f = stripAccents(values.competition).toLowerCase();
  1273→        if (/\bfoot/.test(f)) values.sport = "FOOTBALL";
  1274→        else if (/\bbasket/.test(f)) values.sport = "BASKETBALL";
  1275→        else if (/\btennis/.test(f)) values.sport = "TENNIS";
  1276→        if (values.sport) filled.push("sport (repli)");
  1277→      }
  1278→    }
  1279→    if (!values.matchDate) {
  1280→      const v = findL(/^(date( et heure| du match)?|heure|jour)$/);
  1281→      if (v) {
  1282→        const local = parseMatchDate(v, warnings);
  1283→        if (local) {
  1284→          values.matchDate = local;
  1285→          filled.push("date du match (repli)");
  1286→          warnings.push("date/heure importée — à contrôler");
  1287→        }
  1288→      }
  1289→    }
  1290→    if (!values.market) {
  1291→      const v = findL(/^(marche|market|pari retenu)$/);
  1292→      if (v) {
  1293→        values.market = v.split(/\s+\(ou\b/i)[0].replace(/[;.]+$/, "").trim().slice(0, 120) || undefined;
  1294→        if (values.market) filled.push("marché (repli)");
  1295→      }
  1296→    }
