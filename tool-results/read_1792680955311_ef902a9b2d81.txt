   274→   274→function splitSentences(text: string): string[] {
   275→   275→  return text
   276→   276→    .replace(/\s+/g, " ")
   277→   277→    .split(/(?<=[.!?])\s+/)
   278→   278→    .map((s) => s.trim())
   279→   279→    .filter(Boolean);

   282→   282→/** Premières phrases d'un passage (résume une puce en quelques mots),
   283→   283→    coupé en fin de MOT si la limite est dépassée (jamais au milieu). */
   284→   284→function firstSentences(text: string, maxChars: number, maxCount = 2): string {
   285→   285→  const sentences = splitSentences(text);
   286→   286→  if (!sentences.length) return "";
   287→   287→  const out: string[] = [];
   288→   288→  for (const s of sentences) {
   289→   289→    if (out.length >= maxCount) break;
   290→   290→    if (out.length > 0 && out.join(" ").length + s.length + 1 > maxChars) break;
   291→   291→    out.push(s);
   292→   292→  }
   293→   293→  const joined = out.join(" ");
   294→   294→  if (joined.length > maxChars) {
   295→   295→    const cut = joined.slice(0, maxChars);
   296→   296→    const sp = cut.lastIndexOf(" ");
   297→   297→    return (sp > maxChars * 0.6 ? cut.slice(0, sp) : cut).trimEnd() + " …";
   298→   298→  }
   299→   299→  return joined;
   300→   300→}

   303→   303→function truncateAtSentence(text: string, max: number): string {
   304→   304→  if (text.length <= max) return text;
   305→   305→  const cut = text.slice(0, max);
   306→   306→  const m = cut.match(/^[\s\S]*[.!?](?=\s|$)/);
   307→   307→  if (m && m[0].trim().length > max * 0.5) return m[0].trim() + " …";
   308→   308→  return cut.trimEnd() + " …";
   309→   309→}

   311→   311→/** Supprime les doublons (B et F répétant la même idée) par empreinte. */
   312→   312→function dedupeItems(items: string[]): string[] {
   313→   313→  const seen = new Set<string>();
   314→   314→  const out: string[] = [];
   315→   315→  for (const it of items) {
   316→   316→    const key = stripAccents(it).toLowerCase().replace(/[^a-z0-9]+/g, " ").trim().slice(0, 60);
   317→   317→    if (!key || seen.has(key)) continue;
   318→   318→    seen.add(key);
   319→   319→    out.push(it.trim());
   320→   320→  }
   321→   321→  return out;
   322→   322→}

   326→   326→// Mots-clés de titres de sections, dans l'ordre canonique A → G.

   338→   338→  [new RegExp(ARTICLE_PREFIX + "(?:verdict|decision\\s+finale)"), "F"],
   339→   339→  [new RegExp(ARTICLE_PREFIX + "(synthese|conclusion|decision\\s+finale)"), "G"],

   343→   343→  "match et contexte",
   344→   344→  "match & contexte",

   346→   346→  "scenario sportif",
   347→   347→  "convergence des preuves",

   350→   350→  "comparaison",

   357→   357→function isSectionTitle(text: string): boolean {
   358→   358→  const t = stripAccents(text)
   359→   359→    .toLowerCase()
   360→   360→    .replace(/[^a-z& ]+/g, " ")
   361→   361→    .replace(/\s+/g, " ")
   362→   362→    .replace(/^(?:le|la|les|notre|nos)\s+/, "")
   363→   363→    .trim();
   364→   364→  return SECTION_KEYWORDS.some((k) => t.startsWith(k));

   367→   367→/** Découpe le texte en sections nommées (A à H). Accepte :
   368→   368→ *  - lettres « A. / B) / C - » (majuscules ou minuscules, avec ou sans gras)
   369→   369→ *  - numérotation quelconque « 1. 2. » « I. II. » si le titre est reconnu
   370→   370→ *  - titres nus « MATCH ET CONTEXTE » sans aucun préfixe
   371→   371→ *  Sans en-têtes reconnaissables → tout dans "ALL" (remplissage par étiquettes). */
   372→   372→function splitSections(text: string): Record<string, string> {
   373→   373→  const lines = cleanText(text).split("\n");
   374→   374→  const sections: Record<string, string[]> = { ALL: [] };
   375→   375→  let current = "ALL";
   376→   376→  const headerRe = /^\s*([A-Ha-h])\s*[.)\]:\-–—]\s*(.+?)\s*$/;
   377→   377→  const bulletRe = /^\s*[•‣▪◦*\-–—]\s+/;
   378→   378→  for (const line of lines) {
   379→   379→    const stripped = stripInline(line);
   380→   380→    if (!stripped || DECOR_LINE_RE.test(stripped)) continue; // vide / décoratif
   381→   381→    const m = stripped.match(headerRe);
   382→   382→    const letters = m ? m[2].replace(/[^A-Za-zÀ-ÿ]/g, "") : "";
   383→   383→    const isHeader =
   384→   384→      !!m && letters.length > 2 && (letters === letters.toUpperCase() || isSectionTitle(m[2]));
   385→   385→    if (isHeader && m) {
   386→   386→      current = m[1].toUpperCase();
   387→   387→      if (!sections[current]) sections[current] = [];
   388→   388→      continue;
   389→   389→    }
   390→   390→    // En-tête sans lettre reconnue (numéroté ou nu) : titre court, pas de puce,
   391→   391→    // pas de « : » — la position dans l'ordre canonique tranche la section.
   392→   392→    // Pour une ligne NUMÉROTÉE (« 2. Convergence… »), le titre doit rester très
   393→   393→    // court : au-delà, c'est presque toujours une raison/liste, pas un titre.
   394→   394→    const numberedRe = /^\s*(?:\d{1,2}|[ivxlIVXL]{1,5})\s*[.)\]:\-–—]/;
   395→   395→    const maxTitle = numberedRe.test(stripped) ? 45 : 80;
   396→   396→    if (!bulletRe.test(stripped) && !/:/.test(stripped) && stripped.length < maxTitle) {
   397→   397→      const t = stripAccents(stripped)
   398→   398→        .toLowerCase()
   399→   399→        .replace(/^\s*(?:\d{1,2}|[ivxlIVXL]{1,5})\s*[.)\]:\-–—]?\s*/, "")
   400→   400→        .replace(/\s+/g, " ")
   401→   401→        .trim();
   402→   402→      const hit = ORDER_KEYWORDS.find(([re]) => re.test(t));
   403→   403→      if (hit) {
   404→   404→        current = hit[1];
   405→   405→        if (!sections[current]) sections[current] = [];
   406→   406→        continue;
   407→   407→      }
   408→   408→    }
   409→   409→    sections[current].push(stripped);
   410→   410→  }
   411→   411→  const out: Record<string, string> = {};
   412→   412→  for (const [k, v] of Object.entries(sections)) out[k] = v.join("\n").trim();
   413→   413→  return out;
   414→   414→}

   416→   416→/** Découpe une section en puces « Label : corps » (•, -, –, —, *) ; les lignes
   417→   417→    simples sont rattachées à la puce précédente (continuation / sous-listes). */
   418→   418→function parseBullets(section: string): { label: string; body: string }[] {
   419→   419→  const lines = cleanText(section).split("\n");
   420→   420→  const bullets: { label: string; body: string }[] = [];
   421→   421→  const bulletRe = /^\s*[•‣▪◦*\-–—]\s+(.*)$/;
   422→   422→  for (const line of lines) {
   423→   423→    if (DECOR_LINE_RE.test(line.trim())) continue; // séparateur décoratif
   424→   424→    const m = line.match(bulletRe);
   425→   425→    if (m) {
   426→   426→      const t = m[1].trim();
   427→   427→      const ci = t.indexOf(":");
   428→   428→      if (ci > 0 && ci < 80) {
   429→   429→        bullets.push({ label: t.slice(0, ci).trim(), body: t.slice(ci + 1).trim() });
   430→   430→      } else {
   431→   431→        bullets.push({ label: "", body: t });
   432→   432→      }
   433→   433→    } else if (line.trim()) {
   434→   434→      const t = line.trim();
   435→   435→      // Ligne « Label : valeur » SANS puce (collage sans mise en forme) :
   436→   436→      // traitée comme une puce étiquetée plutôt que comme du prose perdue.
   437→   437→      const lm = t.match(LABEL_LINE_RE);
   438→   438→      const pm = lm ? null : t.match(PROSE_LABEL_RE);
   439→   439→      if (lm) {
   440→   440→        bullets.push({ label: lm[1].trim(), body: lm[2].replace(STOP_LABEL_RE, "").trim() });
   441→   441→      } else if (pm) {
   442→   442→        bullets.push({ label: t.slice(0, t.indexOf(":")).trim(), body: pm[1].replace(STOP_LABEL_RE, "").trim() });
   443→   443→      } else if (bullets.length && !isShortLabel(bullets[bullets.length - 1].label)) {
   444→   444→        bullets[bullets.length - 1].body += `\n${t}`;
   445→   445→      } else if (!bullets.length) {
   446→   446→        bullets.push({ label: "", body: t });
   447→   447→      }
   448→   448→    }
   449→   449→  }
   450→   450→  return bullets;
   451→   451→}

   465→   465→    .split("\n")
   466→   466→    .map((l) => l.replace(/^\s*[-–—•*]\s*/, "").replace(/^\d+\s*[.)]\s*/, "").trim())
   467→   467→    .filter(Boolean);
   468→   468→  if (lines.length > 1) return lines;

   478→   478→    .split(/(?<=[.!?])\s+/)
   479→   479→    .map((s) => s.trim())
   480→   480→    .filter((s) => s.length >= 24);
   481→   481→  return sentences.length > 1 ? sentences : [flat];
   482→   482→}

   514→   514→/* ---------- Date & heure ---------- */
   515→   515→
   516→   516→function parseMatchDate(body: string, warnings: string[]): string | undefined {
   517→   517→  const flat = body.replace(/\n/g, " ");
   518→   518→
   519→   519→  // Format littéraire : « 17 octobre 2026 » / « 3 novembre 2025 »
   520→   520→  let day: number | undefined;
   521→   521→  let month: number | undefined;
   522→   522→  let year: number | undefined;
   523→   523→  let dateIdx = -1;
   524→   524→  const dm = flat.match(new RegExp(`(\\d{1,2})(?:er)?\\s+(${MONTH_KEYS})\\s+(\\d{4})`, "i"));
   525→   525→  if (dm) {
   526→   526→    dateIdx = dm.index ?? -1;
   527→   527→    day = parseInt(dm[1], 10);
   528→   528→    month = MONTHS[dm[2].toLowerCase()];
   529→   529→    year = parseInt(dm[3], 10);
   530→   530→  } else {
   531→   531→    // Formats numériques : 17/10/2026, 17-10-2026, 17.10.2026, 2026-10-17
   532→   532→    const ymd = flat.match(/\b(\d{4})[-/](\d{1,2})[-/](\d{1,2})\b/);
   533→   533→    const dmy = ymd ? undefined : flat.match(/\b(\d{1,2})[/.‐-](\d{1,2})[/.‐-](\d{4})\b/);
   534→   534→    const parts = ymd ?? dmy;
   535→   535→    if (parts) {
   536→   536→      dateIdx = parts.index ?? -1;
   537→   537→      const a = parseInt(parts[1], 10);
   538→   538→      const b = parseInt(parts[2], 10);
   539→   539→      const c = parseInt(parts[3], 10);
   540→   540→      if (ymd) {
   541→   541→        year = a; month = b; day = c;
   542→   542→      } else {
   543→   543→        // Ambiguïté jour/mois : la valeur > 12 désigne forcément le jour
   544→   544→        day = a > 12 ? a : b > 12 ? b : a;
   545→   545→        month = day === a ? b : a;
   546→   546→        year = c;
   547→   547→      }
   548→   548→    }
   549→   549→  }
   550→   550→  if (day === undefined || month === undefined || year === undefined) return undefined;
   551→   551→  if (month < 1 || month > 12 || day < 1 || day > 31) return undefined;
   552→   552→
   553→   553→  let hour = 12;
   554→   554→  let minute = 0;
   555→   555→  // L'heure est cherchée AU VOISINAGE de la date, jamais n'importe où dans
   556→   556→  // le texte : « 21h00 » collé à la date, pas « 8 buts » ailleurs.
   557→   557→  // « h » non suivi d'une lettre (« 2 heures » ≠ 2 h du matin).
   558→   558→  const TIME_RE = /(\d{1,2})\s*(?::\s*(\d{1,2})|h(?![a-zà-ÿ])\s*(\d{1,2})?)/i;
   559→   559→  const near = dateIdx >= 0 ? flat.slice(Math.max(0, dateIdx - 24), dateIdx + 90) : "";
   560→   560→  const tm = (dateIdx >= 0 ? near.match(TIME_RE) : null) ?? flat.match(TIME_RE);
   561→   561→  let hasTime = false;
   562→   562→  if (tm) {
   563→   563→    hour = parseInt(tm[1], 10);
   564→   564→    minute = parseInt(tm[2] ?? tm[3] ?? "0", 10);
   565→   565→    hasTime = true;
   566→   566→  }
   567→   567→
   568→   568→  const tzm = flat.match(
   569→   569→    /\b(CEST|CETE?|WEST|WET|EEST|EET|PDT|PST|MDT|MST|CDT|CST|EDT|EST|UTC|GMT)\b/i
   570→   570→  );
   571→   571→  let offset = 0;
   572→   572→  if (tzm) {
   573→   573→    offset = TZ_OFFSETS[tzm[1].toUpperCase()] ?? 0;
   574→   574→  } else {
   575→   575→    warnings.push("fuseau horaire non détecté — heure lue comme UTC, vérifiez la date");
   576→   576→  }
   577→   577→  if (!hasTime) {
   578→   578→    warnings.push("heure du match non détectée — 12 h placée par défaut");
   579→   579→  }
   580→   580→
   581→   581→  const utcMs = Date.UTC(year, month - 1, day, hour, minute) - offset * 3_600_000;
   582→   582→  return toDatetimeLocal(new Date(utcMs));
   583→   583→}

   585→   585→/* ---------- Statistiques clés (section C) ---------- */
   586→   586→
   587→   587→/**
   588→   588→ * Extraction prudente : seules les statistiques « propres » sont conservées
   589→   589→ * (étiquette courte et lisible devant le nombre, ou étiquette après « : »).
   590→   590→ * Motifs reconnus : bilans de forme « NV-ND-ND » (1V-2N-5D), rapports « 9/10 »,
   591→   591→ * scores « N-N » (H2H 6-1), et nombre + unité (% , buts, matchs, sets…).
   592→   592→ * Les fragments de prose sont ignorés (comptés dans `skipped`) afin de ne pas
   593→   593→ * polluer le public — le texte brut reste disponible dans le scénario.
   594→   594→ */
   595→   595→function parseKeyStats(body: string): { stats: KeyStatItem[]; skipped: number } {
   596→   596→  // Protège les décimales (2,33) avant découpe sur les virgules
   597→   597→  const flat = body.replace(/\s+/g, " ").replace(/(\d),(\d)/g, "$1\u0001$2");
   598→   598→  const clauses = flat
   599→   599→    .split(/\s*[;.]\s+|\s*[;]\s*|\s*,\s+/)
   600→   600→    .map((c) => c.replace(/\u0001/g, ",").trim())
   601→   601→    .filter((c) => stripAccents(c).length >= 6);
   602→   602→
   603→   603→  const stats: KeyStatItem[] = [];
   604→   604→  let skipped = 0;
   605→   605→
   606→   606→  /** Nettoie puis valide une étiquette de stat (la prose est rejetée). */
   607→   607→  const makeLabel = (raw: string): string | null => {
   608→   608→    const label = raw
   609→   609→      .replace(/[~*]+$/, "")
   610→   610→      .replace(/^[\s~*>•\-–—]+/, "")
   611→   611→      .replace(/[\s:;,–—~*\-]+$/, "")
   612→   612→      .replace(/\s+(dans|de|du|des|la|le|les|un|une|en|sur|par|pour|avec|et|ou|au|aux|vs|chez|face|contre)\s*$/i, "")
   613→   613→      .replace(/[\s:;,–—~*\-]+$/, "")
   614→   614→      .trim();
   615→   615→    if (label.length < 2 || label.length > 48) return null;
   616→   616→    if (/[«»"'()[\]]/.test(label)) return null;
   617→   617→    if (!/^[A-Za-zÀ-ÿ]/.test(label)) return null;
   618→   618→    if (JUNK_LABELS.has(stripAccents(label).toLowerCase())) return null;
   619→   619→    return label;
   620→   620→  };
   621→   621→
   622→   622→  const pushStat = (rawLabel: string, value: string, detail?: string): boolean => {
   623→   623→    const label = makeLabel(rawLabel);
   624→   624→    if (!label) {
   625→   625→      skipped++;
   626→   626→      return false;
   627→   627→    }
   628→   628→    stats.push({
   629→   629→      label: label.slice(0, 120),
   630→   630→      value: value.slice(0, 60),
   631→   631→      ...(detail && detail !== label ? { detail: detail.slice(0, 300) } : {}),
   632→   632→    });
   633→   633→    return true;
   634→   634→  };
   635→   635→
   636→   636→  for (const clause of clauses) {
   637→   637→    if (stats.length >= MAX_POINTS) break;
   638→   638→    const ci = clause.indexOf(":");
   639→   639→    const base = ci > 0 ? clause.slice(ci + 1) : clause;
   640→   640→
   641→   641→    /* Bilan de forme « NV-ND-ND » (1V-2N-5D…) : motif dédié, AVANT les
   642→   642→       scores, car « 1V-2N-5D » ne doit pas être lu comme « 5-D ». */
   643→   643→    const recRe = /(?<![\w\d.])(\d{1,2})\s*V\s*-\s*(\d{1,2})\s*N\s*-\s*(\d{1,2})\s*D(?![\w])/i;
   644→   644→    const rm = base.match(recRe);
   645→   645→    if (rm && rm.index !== undefined) {
   646→   646→      const before = base.slice(0, rm.index).trim();
   647→   647→      if (before.length >= 2 && !before.includes("(") && /[A-Za-zÀ-ÿ]/.test(before)) {
   648→   648→        const detail = base
   649→   649→          .slice(rm.index + rm[0].length)
   650→   650→          .trim()
   651→   651→          .replace(/^[\s,%–—/-]+/, "")
   652→   652→          .split("(")[0]
   653→   653→          .trim();
   654→   654→        if (pushStat(before.replace(/[\s:;,–—-]+$/, ""), `${rm[1]}V-${rm[2]}N-${rm[3]}D`, detail)) {
   655→   655→          continue;
   656→   656→        }
   657→   657→      } else {
   658→   658→        skipped++;
   659→   659→        continue;
   660→   660→      }
   661→   661→    }
   662→   662→
   663→   663→    /* Rapport « 9/10 » (forme récente) : dédié, avant les scores ; les
   664→   664→       gardes excluent les dates (17/10/2026) et les cotes « 2,25/2,5 ». */
   665→   665→    const ratioRe = /(?<![\d.,%/-])(\d{1,3})\s*\/\s*(\d{1,3})(?![\d.,%/-])/;
   666→   666→    const tm = base.match(ratioRe);
   667→   667→    if (tm && tm.index !== undefined) {
   668→   668→      const before = base.slice(0, tm.index).trim();
   669→   669→      if (before.length >= 2 && !before.includes("(") && /[A-Za-zÀ-ÿ]/.test(before)) {
   670→   670→        const detail = base
   671→   671→          .slice(tm.index + tm[0].length)
   672→   672→          .trim()
   673→   673→          .replace(/^[\s,%–—/-]+/, "")
   674→   674→          .split("(")[0]
   675→   675→          .trim();
   676→   676→        if (pushStat(before.replace(/[\s:;,–—-]+$/, ""), `${tm[1]}/${tm[2]}`, detail)) {
   677→   677→          continue;
   678→   678→        }
   679→   679→      } else {
   680→   680→        skipped++;
   681→   681→        continue;
   682→   682→      }
   683→   683→    }
   684→   684→
   685→   685→    /* Score « N-N » (H2H 6-1, bilan 3-2…) : motif dédié, AVANT le nombre
   686→   686→       générique, pour éviter la lecture partielle « 6 ». Gardes anti-faux
   687→   687→       positifs : pas de chiffre/point/virgule/% collé autour (exclut les
   688→   688→       plages décimales « 0,98-1,13 »). */
   689→   689→    const scoreRe = /(?<![\d.,%])(\d{1,3})\s*[-–]\s*(\d{1,3})(?![\d.,%])/;
   690→   690→    const sm = base.match(scoreRe);
   691→   691→    if (sm && sm.index !== undefined) {
   692→   692→      const before = base.slice(0, sm.index).trim();
   693→   693→      if (
   694→   694→        before.length >= 2 && before.length <= 48 &&
   695→   695→        !before.includes("(") && /[A-Za-zÀ-ÿ]/.test(before)
   696→   696→      ) {
   697→   697→        const after = base.slice(sm.index + sm[0].length).trim();
   698→   698→        const detail = after.replace(/^[\s,%–—/-]+/, "").split("(")[0].trim();
   699→   699→        if (pushStat(before.replace(/[\s:;,–—-]+$/, ""), `${sm[1]}-${sm[2]}`, detail)) {
   700→   700→          continue;
   701→   701→        }
   702→   702→      } else {
   703→   703→        skipped++;
   704→   704→        continue;
   705→   705→      }
   706→   706→    }
   707→   707→
   708→   708→    const numRe = /([+\-−~]?\s?\d[\d\u0001\u202f ]*(?:[.,]\d+)?)/;
   709→   709→    const nm = base.match(numRe);
   710→   710→    if (!nm) continue;
   711→   711→    const after = base.slice((nm.index ?? 0) + nm[0].length).trim();
   712→   712→    const unit = after.match(/^(%|u\b|buts?|jeux?|points?|pts?\b|match(?:e)?s?|set|sets|aces?|fautes?)/i);
   713→   713→    const unitTxt = unit ? unit[0] : "";
   714→   714→    const numStr = nm[0].replace(/\u0001/g, ",").replace(/\s{2,}/g, " ").trim();
   715→   715→
   716→   716→    // Année isolée (1900-2099) sans unité → pas une statistique
   717→   717→    if (!unitTxt && /^(19|20)\d{2}$/.test(numStr)) {
   718→   718→      skipped++;
   719→   719→      continue;
   720→   720→    }
   721→   721→
   722→   722→    let label: string;
   723→   723→    if (ci > 0) {
   724→   724→      label = clause.slice(0, ci).trim();
   725→   725→      if (label.length > 64 || !/[A-Za-zÀ-ÿ]/.test(label)) {
   726→   726→        skipped++;
   727→   727→        continue;
   728→   728→      }
   729→   729→    } else {
   730→   730→      const before = base.slice(0, nm.index ?? 0).trim();
   731→   731→      if (before.length < 2 || before.length > 48 || before.includes("(") || !/[A-Za-zÀ-ÿ]/.test(before)) {
   732→   732→        skipped++;
   733→   733→        continue;
   734→   734→      }
   735→   735→      label = before;
   736→   736→    }
   737→   737→
   738→   738→    const cleanLabel = makeLabel(label);
   739→   739→    if (!cleanLabel) {
   740→   740→      skipped++;
   741→   741→      continue;
   742→   742→    }
   743→   743→
   744→   744→    const detail = (unitTxt ? after.slice(unitTxt.length) : after).replace(/^[\s,%–—/-]+/, "").trim();
   745→   745→    stats.push({
   746→   746→      label: cleanLabel.slice(0, 120),
   747→   747→      value: (numStr + (unitTxt ? ` ${unitTxt}` : "")).slice(0, 60),
   748→   748→      ...(detail && detail !== cleanLabel ? { detail: detail.slice(0, 300) } : {}),
   749→   749→    });
   750→   750→  }
   751→   751→  return { stats, skipped };
   752→   752→}
   753→   753→
   754→   754→/* ---------- Niveau de confiance ---------- */

   756→   756→function levelToConfidence(level: string): string {
   757→   757→  if (level.startsWith("tr")) return "TRES_ELEVE";
   758→   758→  if (level.startsWith("faible")) return "FAIBLE";
   759→   759→  if (level.startsWith("modere")) return "MODERE";
   760→   760→  return "ELEVE";
   761→   761→}
   762→   762→

   765→   765→/* ---------- Parseur principal ---------- */
   766→   766→
   767→   767→export function parseAnalyzeText(raw: string): ImportReport {
   768→   768→  const warnings: string[] = [];
   769→   769→  const parseErrors: string[] = [];
   770→   770→  const values: ImportedValues = {};
   771→   771→  const filled: string[] = [];
   772→   772→  const synthesisBlocks: string[] = [];
   773→   773→  let contextLine = "";
   774→   774→  // Texte complet aplati + détecteur de puces (pour le repli global)
   775→   775→  const flatRawAll = cleanText(raw).replace(/\s+/g, " ");
   776→   776→  const bulletReTop = /^\s*[•‣▪◦*\-–—]\s+/;
   777→   777→
   778→   778→  if (!raw || stripAccents(raw).replace(/\W/g, "").length < 40) {
   779→   779→    return { values, filled, warnings, parseErrors: ["Texte trop court ou vide."] };
   780→   780→  }
   781→   781→
   782→   782→  const sections = splitSections(raw);
   783→   783→  const hasSections = Object.keys(sections).some((k) => k !== "ALL");
   784→   784→  if (!hasSections) {
   785→   785→    warnings.push("sections A à G non repérées — remplissage partiel par étiquettes");
   786→   786→  }
   787→   787→  const sec = (k: string) => sections[k] ?? (hasSections ? "" : sections.ALL ?? "");
   788→   788→  const bulletsOf = (k: string) => parseBullets(sec(k));
   789→   789→
   790→   790→  /* ---- Titre (secours pour l'affiche) ---- */
   791→   791→  const titleLine = cleanText(raw).match(/Analyse\s*:\s*(.+)$/im);
   792→   792→  if (titleLine) {
   793→   793→    const title = titleLine[1].trim();
   794→   794→    if (title.length >= 3 && title.length <= 160) {
   795→   795→      values.matchName = title;
   796→   796→      filled.push("affiche (titre)");
   797→   797→    }
   798→   798→  }
   799→   799→
   800→   800→  /* ---- A. Match et contexte ---- */
   801→   801→  const aBullets = bulletsOf("A");
   802→   802→  if (aBullets.length) {
   803→   803→    const matchB = findBullet(aBullets, /^(match|rencontre|affiche|evenement)\b/);
   804→   804→    if (matchB?.body) {
   805→   805→      values.matchName = matchB.body.replace(/\s+/g, " ").trim().replace(/[.]\s*$/, "").slice(0, 160);
   806→   806→      filled.push("affiche");
   807→   807→    }
   808→   808→    const sportB = findBullet(aBullets, /^sport/);
   809→   809→    const compB = findBullet(aBullets, /^competition/);
   810→   810→    // La compétition peut être collée à la ligne « Sport » OU avoir sa propre ligne :
   811→   811→    // les deux sources sont examinées, indépendamment l'une de l'autre.
   812→   812→    const compSource = sportB?.body || compB?.body || "";
   813→   813→    if (sportB?.body || compB?.body) {
   814→   814→      const flat = stripAccents(`${sportB?.body ?? ""} ${compB?.body ?? ""}`).toLowerCase();
   815→   815→      if (/\bfootball\b|\bsoccer\b|\bfoot\b/.test(flat)) values.sport = "FOOTBALL";
   816→   816→      else if (/\bbasket/.test(flat)) values.sport = "BASKETBALL";
   817→   817→      else if (/\btennis\b/.test(flat)) values.sport = "TENNIS";
   818→   818→
   819→   819→      const comp =
   820→   820→        (compB?.body ? extractCompetition(compB.body) : {});
   821→   821→      // Repli sur la ligne « Sport » si la ligne « Compétition » n'a rien donné
   822→   822→      if (!comp.competition && sportB?.body) {
   823→   823→        const c2 = extractCompetition(sportB.body);
   824→   824→        if (c2.competition) (comp as { competition?: string }).competition = c2.competition;
   825→   825→        if (c2.sportHint && !comp.sportHint) (comp as { sportHint?: string }).sportHint = c2.sportHint;
   826→   826→      }
   827→   827→      if (comp.sportHint && !values.sport) values.sport = comp.sportHint;
   828→   828→      if (values.sport) filled.push("sport");
   829→   829→      else warnings.push("sport non reconnu — laissez la valeur actuelle");
   830→   830→
   831→   831→      if (comp.competition) {
   832→   832→        values.competition = comp.competition.slice(0, 120);
   833→   833→        filled.push("compétition");
   834→   834→      } else if (compB?.body) {
   835→   835→        const label = cleanCompetitionLabel(compB.body);
   836→   836→        if (label) {
   837→   837→          values.competition = label.slice(0, 120);
   838→   838→          filled.push("compétition");
   839→   839→        }
   840→   840→      }
   841→   841→      if (!values.competition && hasSections) {
   842→   842→        warnings.push("compétition introuvable dans la section A — à compléter");
   843→   843→      }
   844→   844→    }
   845→   845→    const dateB = findBullet(aBullets, /^date/);
   846→   846→    if (dateB?.body) {
   847→   847→      const local = parseMatchDate(dateB.body, warnings);
   848→   848→      if (local) {
   849→   849→        values.matchDate = local;
   850→   850→        filled.push("date du match");
   851→   851→        warnings.push("date/heure importée — à contrôler");
   852→   852→        const dateRe2 = new RegExp(`^\\d{1,2}(?:er)?\\s+(${MONTH_KEYS})\\s+\\d{4}$`, "i");
   853→   853→        const weekdayRe = /^(lundi|mardi|mercredi|jeudi|vendredi|samedi|dimanche)$/i;
   854→   854→        // Retire les parenthèses contenant une heure (ex. « (autour de 08:30–10:30 …) »)
   855→   855→        const cleaned = dateB.body.replace(/\([^()]*\d{1,2}\s*[h:][^()]*\)/g, " ");
   856→   856→        const rest = cleaned
   857→   857→          .split(",")
   858→   858→          .map((s) => s.trim())
   859→   859→          .filter((s) => {
   860→   860→            if (!s) return false;
   861→   861→            const t = s.replace(/\s+/g, " ").trim();
   862→   862→            const stripped = t.replace(/^(lundi|mardi|mercredi|jeudi|vendredi|samedi|dimanche)\s+/i, "");
   863→   863→            if (weekdayRe.test(t)) return false;
   864→   864→            if (dateRe2.test(stripped)) return false;
   865→   865→            // Tout résidu lié à l'horaire est du bruit : heure « 21h00 »,
   866→   866→            // plage « 20h00-21h00 », fuseau « CET/CEST/UTC… », indications
   867→   867→            // de « autour de 20h » / « heure locale » — jamais du contexte utile.
   868→   868→            if (/\d{1,2}\s*[h:]/i.test(t)) return false;
   869→   869→            if (/\b(?:CEST|CET|WEST|WET|UTC|GMT|EET|EST|PST|PDT|CDT|CST|MDT|MST)\b/i.test(t)) return false;
   870→   870→            if (/(autour de|environ|heure locale|locale|fuseaux?|horaire)/i.test(t)) return false;
   871→   871→            // Seul un libellé significatif SANS chiffre mérite d'être conservé
   872→   872→            // (ex. « Mestalla », « Estádio do Dragão »).
   873→   873→            if (/\d/.test(t)) return false;
   874→   874→            if (stripAccents(t).replace(/[^A-Za-z]/g, "").length < 4) return false;
   875→   875→            return true;
   876→   876→          })
   877→   877→          .join(", ")
   878→   878→          .replace(/[()\[\]]/g, "")
   879→   879→          .replace(/\s{2,}/g, " ")
   880→   880→          .trim();
   881→   881→        if (rest && rest.length > 3) contextLine = rest.slice(0, 300);
   882→   882→      }
   883→   883→    } else if (hasSections) {
   884→   884→      warnings.push("date du match introuvable dans la section A");
   885→   885→    }
   886→   886→  }
   887→   887→
   888→   888→  /* ---- B. Scénario sportif ---- */
   889→   889→  const bBullets = bulletsOf("B");
   890→   890→  const keyPointItems: string[] = [];
   891→   891→  let supLine = ""; // « Équipe supérieure : … » / « Joueur supérieur : … »
   892→   892→  let attenduLine = ""; // « Scénario attendu : … » (cœur du scénario)
   893→   893→  let raisonsLine = ""; // premières raisons (pour le résumé)
   894→   894→  const bSubReasons: string[] = []; // sous-puces de raisons sans étiquette
   895→   895→  const extraBLines: string[] = []; // autres puces étiquetées (1 phrase max)
   896→   896→
   897→   897→  if (bBullets.length) {
   898→   898→    for (const b of bBullets) {
   899→   899→      const lab = normLabel(b.label);
   900→   900→      if (/^raisons principales/.test(lab)) {
   901→   901→        keyPointItems.push(...splitReasons(b.body));
   902→   902→        raisonsLine = firstSentences(b.body, 300, 2);
   903→   903→      } else if (/^scenario alternatif|^alternatif/.test(lab)) {
   904→   904→        if (b.body) {
   905→   905→          values.alternativeScenario = b.body.replace(/\s+/g, " ").trim().slice(0, 2000);
   906→   906→          filled.push("scénario alternatif");
   907→   907→        }
   908→   908→      } else if (/^facteurs/.test(lab)) {
   909→   909→        const items = b.body
   910→   910→          .split(/\s*;\s*|\n/)
   911→   911→          .map((l) => l.replace(/^\s*[-–—•*]\s*/, "").replace(/^\d+\s*[.)]\s*/, "").trim())
   912→   912→          .filter((l) => l.length >= 6);
   913→   913→        keyPointItems.push(...items);
   914→   914→      } else if (hasSections && !b.label && b.body && b.body.replace(/\W/g, "").length >= 12) {
   915→   915→        // Sous-puces imbriqués (ex. liste de raisons sans étiquette) —
   916→   916→        // uniquement en mode sections : en collage plat, verser toutes les
   917→   917→        // lignes de prose dans les points clés « ramassait tout ».
   918→   918→        const items = b.body
   919→   919→          .split("\n")
   920→   920→          .map((l) => l.replace(/^\s*[-–—•*]\s*/, "").replace(/^\d+\s*[.)]\s*/, "").trim())
   921→   921→          .filter((l) => l.length >= 6);
   922→   922→        keyPointItems.push(...items);
   923→   923→        bSubReasons.push(...items.slice(0, 2));
   924→   924→      } else if (/^(equipe|joueur)\s+superieur/.test(lab) && b.body) {
   925→   925→        supLine = `${b.label} : ${b.body.replace(/\s+/g, " ").trim().replace(/[.]\s*$/, "")}`;
   926→   926→      } else if (/^(scenario\s+attendu|attendu)$/.test(lab) && b.body) {
   927→   927→        attenduLine = `Scénario attendu : ${firstSentences(b.body, 320, 2)}`;
   928→   928→      } else if (b.label && b.body && b.body.replace(/\W/g, "").length >= 12) {
   929→   929→        // Autre puce étiquetée : une seule phrase au résumé (jamais tout).
   930→   930→        extraBLines.push(`${b.label} : ${firstSentences(b.body, 160, 1)}`);
   931→   931→      }
   932→   932→      // Les puces sans étiquette en collage plat ne vont nulle part ici :
   933→   933→      // la prose narrative est gérée par l'assemblage du scénario.
   934→   934→    }
   935→   935→  }
   936→   936→
   937→   937→  /* ---- C. Convergence des preuves ----
   938→   938→     TOUTES les puces de C passent au détecteur de statistiques (pas
   939→   939→     seulement « Statistiques : ») : bilans NV-ND-ND, scores, rapports,
   940→   940→     %, volumes avec unité… Les puces marché/cotes sont écartées (leurs
   941→   941→     valeurs alimentent les champs E). Une puce SANS statistique lisible
   942→   942→     devient un POINT CLÉ de décision (blessures, modèles, tactique…)
   943→   943→     au lieu d'être perdue. Un paragraphe libre de C devient la phrase
   944→   944→     « Convergence : » du résumé. */
   945→   945→  const cBullets = bulletsOf("C");
   946→   946→  const statAcc: KeyStatItem[] = [];
   947→   947→  const seenStats = new Set<string>();
   948→   948→  let cLead = "";
   949→   949→  let statBulletSkipped = 0;
   950→   950→  if (hasSections) {
   951→   951→    // Paragraphe de convergence NON à puces (synthèse libre de la section C)
   952→   952→    const leadParts = sec("C")
   953→   953→      .split("\n")
   954→   954→      .map((l) => l.trim())
   955→   955→      .filter((l) => l && !/^\s*[•‣▪◦*\-–—]/.test(l) && !LABEL_LINE_RE.test(l) && l.length > 40);
   956→   956→    if (leadParts.length) cLead = leadParts.join(" ").replace(/\s+/g, " ").trim();
   957→   957→
   958→   958→    for (const b of cBullets) {
   959→   959→      const lab = normLabel(b.label);
   960→   960→      const bodyFlat = b.body.replace(/\s+/g, " ").trim();
   961→   961→      const hay = stripAccents(`${b.label} ${bodyFlat}`).toLowerCase();
   962→   962→      const isStatBullet = /^statistiques?/.test(lab);
   963→   963→      if (MARKETISH_RE.test(hay)) continue; // marché/cotes → champs E
   964→   964→      if (bodyFlat) {
   965→   965→        const { stats, skipped } = parseKeyStats(bodyFlat);
   966→   966→        if (isStatBullet) statBulletSkipped = skipped;
   967→   967→        let added = 0;
   968→   968→        for (const s of stats) {
   969→   969→          const key = `${stripAccents(s.label).toLowerCase()}|${s.value}`;
   970→   970→          if (seenStats.has(key)) continue;
   971→   971→          seenStats.add(key);
   972→   972→          if (statAcc.length < MAX_POINTS) {
   973→   973→            statAcc.push(s);
   974→   974→            added++;
   975→   975→          }
   976→   976→        }
   977→   977→        if (added > 0) continue; // la puce vit dans la grille de stats
   978→   978→      }
   979→   979→      // Pas de statistique lisible → point clé de décision (une idée/puce)
   980→   980→      if (bodyFlat.replace(/\W/g, "").length >= 30) {
   981→   981→        const head = b.label ? `${b.label} : ` : "";
   982→   982→        keyPointItems.push(`${head}${firstSentences(bodyFlat, 280, 2)}`.trim());
   983→   983→      }
   984→   984→    }
   985→   985→    if (statAcc.length) {
   986→   986→      values.keyStats = statAcc;
   987→   987→      filled.push(`statistiques clés (${statAcc.length})`);
   988→   988→      if (statBulletSkipped > 0) {
   989→   989→        warnings.push(
   990→   990→          `statistiques partielles (${statBulletSkipped} fragment(s) non décodé(s)) — vérifiez la section C`
   991→   991→        );
   992→   992→      }
   993→   993→    } else if (statBulletSkipped > 0) {
   994→   994→      warnings.push("statistiques non décodées — section C à saisir à la main");
   995→   995→    }
   996→   996→  }
   997→   997→
   998→   998→  /* ---- D. Comparaison des marchés ----
   999→   999→     Détaillée dans les champs et points clés ; volontairement absente du
  1000→  1000→     résumé (sinon le « résumé » retrouve la longueur du texte collé). */
  1001→  1001→
  1002→  1002→  /* ---- Assemblage du scénario principal (un VRAI résumé) ----
  1003→  1003→     Sections reconnues : contexte + équipe supérieure + scénario attendu +
  1004→  1004→     premières raisons + convergence — condensé, coupé en fin de phrase.
  1005→  1005→     La comparaison des marchés (D) et le détail des puces de C restent
  1006→  1006→     hors résumé : ils vivent déjà dans les champs, les stats et les
  1007→  1007→     points clés. Sans sections (collage plat) : narration = tout SAUF
  1008→  1008→     les lignes « Label : valeur » de champs (déjà versées dans le
  1009→  1009→     formulaire) et la ligne d'affiche (déjà dans matchName), condensée
  1010→  1010→     si elle déborde. */
  1011→  1011→  const vsLineRe = /^[A-Za-zÀ-ÿ0-9.'\'\- ]{3,48}\s+vs\.?\s+[A-Za-zÀ-ÿ0-9.'\'\- ]{3,48}\s*$/;
  1012→  1012→  if (!hasSections) {
  1013→  1013→    const fieldLabelsNoNarr = FLAT_LABELS.filter((l) => l !== "equipe superieure");
  1014→  1014→    const fieldLineRe = new RegExp(
  1015→  1015→      `^(?:[-–—•*]\\s*)?(?:${fieldLabelsNoNarr.map((l) => accentPat(l).replace(/ /g, "\\s+")).join("|")})\\s*[:\\-–—]`,
  1016→  1016→      "i"
  1017→  1017→    );
  1018→  1018→    const narrative = cleanText(raw)
  1019→  1019→      .split("\n")
  1020→  1020→      .map((l) => stripInline(l).trim())
  1021→  1021→      .filter(Boolean)
  1022→  1022→      .filter((l) => !fieldLineRe.test(l))

  1357→  1357→       le format A à G : chaque champ critique encore vide est recherché dans
  1358→  1358→       le texte entier (dictionnaire de compétitions, motifs de date, cote,
  1359→  1359→       phrase « je joue … », type de marché). Rien n'est inventé : seul du
  1360→  1360→       texte explicitement présent est utilisé, et la prose n'est jamais
  1361→  1361→       aspirée dans les champs discrets. */
  1362→  1362→    const flatScan = stripAccents(flatRawAll).toLowerCase();
  1363→  1363→
  1364→  1364→    if (!values.sport || !values.competition) {
  1365→  1365→      if (!values.competition) {
  1366→  1366→        for (const c of COMPETITIONS) {
  1367→  1367→          if (c.re.test(flatScan)) {
  1368→  1368→            values.competition = c.label;
  1369→  1369→            filled.push("compétition (scan)");
  1370→  1370→            if (c.sport && !values.sport) {
  1371→  1371→              values.sport = c.sport;
  1372→  1372→              filled.push("sport (scan)");
  1373→  1373→            }
  1374→  1374→            break;
  1375→  1375→          }
  1376→  1376→        }
  1377→  1377→      }
  1378→  1378→      if (!values.sport) {
  1379→  1379→        if (/\bfoot(?:ball)?\b|\bsoccer\b/.test(flatScan)) values.sport = "FOOTBALL";
  1380→  1380→        else if (/\bbasket(?:ball)?\b/.test(flatScan)) values.sport = "BASKETBALL";
  1381→  1381→        else if (/\btennis\b/.test(flatScan)) values.sport = "TENNIS";
  1382→  1382→        if (values.sport) filled.push("sport (scan)");
  1383→  1383→      }
  1384→  1384→    }
  1385→  1385→
  1386→  1386→    if (!values.matchDate) {
  1387→  1387→      const local = parseMatchDate(flatRawAll, warnings);
  1388→  1388→      if (local) {
  1389→  1389→        values.matchDate = local;
  1390→  1390→        filled.push("date du match (scan)");
  1391→  1391→        warnings.push("date du match déduite du texte — à contrôler");
  1392→  1392→      }
  1393→  1393→    }
  1394→  1394→
  1395→  1395→    if (!values.odds) {
  1396→  1396→      const oddsPats: RegExp[] = [
  1397→  1397→        /\bcot(?:e|ation)s?\b[^%\n]{0,24}?(\d{1,2}[.,]\d{1,3})\b/,
  1398→  1398→        /\bodds?\b[^%\n]{0,24}?(\d{1,2}[.,]\d{1,3})\b/,
  1399→  1399→        /@\s*(\d{1,2}[.,]\d{1,3})\b/,
  1400→  1400→      ];
  1401→  1401→      for (const p of oddsPats) {
  1402→  1402→        const m = flatScan.match(p);
  1403→  1403→        if (m) {
  1404→  1404→          const v = Number(m[1].replace(",", "."));
  1405→  1405→          if (v >= 1.01 && v <= 100) {
  1406→  1406→            values.odds = String(v);
  1407→  1407→            filled.push("cote (scan)");
  1408→  1408→            warnings.push("cote importée — à contrôler");
  1409→  1409→            break;
  1410→  1410→          }
  1411→  1411→        }
  1412→  1412→      }
  1413→  1413→    }
  1414→  1414→
  1415→  1415→    if (!values.selection) {
  1416→  1416→      const lineJJ = cleanText(raw)
  1417→  1417→        .split("\n")
  1418→  1418→        .map((l) => stripInline(l).trim())
  1419→  1419→        .find((l) => /je\s+joue/i.test(l));
  1420→  1420→      if (lineJJ) {
  1421→  1421→        const after = lineJJ
  1422→  1422→          .replace(/^.*?je\s+joue(?:\s+sous\s+condition)?\s*(?:[:\-–—]\s*)?/i, "")
  1423→  1423→          .replace(STOP_LABEL_RE, "")
  1424→  1424→          .replace(/\s*@\s*[\d.,]+/gi, "")
  1425→  1425→          .replace(/\s*\(ou\s[^)]*\)\s*$/i, "")
  1426→  1426→          .replace(/\s*[;,.]+\s*$/, "")
  1427→  1427→          .replace(/\s+/g, " ")
  1428→  1428→          .trim();
  1429→  1429→        if (after.length >= 2) {
  1430→  1430→          values.selection = after.slice(0, 120);
  1431→  1431→          filled.push("sélection (scan)");
  1432→  1432→        }
  1433→  1433→      }
  1434→  1434→    }
  1435→  1435→
  1436→  1436→    if (!values.market) {
  1437→  1437→      const hay = `${flatScan} ${values.selection ? stripAccents(values.selection).toLowerCase() : ""}`;
  1438→  1438→      if (/\b(?:over|under|plus de|moins de)\s+\d/.test(hay)) values.market = "Over / Under";
  1439→  1439→      else if (/\basian\s*handicap\b/.test(hay)) values.market = "Asian Handicap";
  1440→  1440→      else if (/\bhandicap/.test(hay)) values.market = "Handicap";
  1441→  1441→      else if (/\bdouble\s+chance\b/.test(hay)) values.market = "Double chance";
  1442→  1442→      else if (/\bbtts\b|deux\s+equipes\s+marquent|les\s+deux\s+marquent/.test(hay))
  1443→  1443→        values.market = "Les deux équipes marquent";
  1444→  1444→      else if (/\bvainqueur\b|\bto\s+win\b|\bwinner\b/.test(hay)) values.market = "Vainqueur du match";
  1445→  1445→      else if (values.selection && /[+\-−]\s?\d+(?:[.,]\d+)?\s*$/.test(values.selection.trim()))
  1446→  1446→        values.market = "Handicap";
  1447→  1447→      if (values.market) filled.push("marché (scan)");
  1448→  1448→    }
  1449→  1449→  }
  1450→  1450→
  1451→  1451→  /* ---- Explicitation des champs restés vides (aucun échec silencieux) ----
  1452→  1452→     Placée en FIN de parseur : elle reflète l'état APRÈS tous les replis
  1453→  1453→     (scan essentiel, ligne dérivée, affiche « vs »). */
  1454→  1454→  if (filled.length >= 2) {
  1455→  1455→    const criticals: [string, string | undefined][] = [
  1456→  1456→      ["affiche (nom du match)", values.matchName],
  1457→  1457→      ["date du match", values.matchDate],
  1458→  1458→      ["marché", values.market],
  1459→  1459→      ["sélection", values.selection],
  1460→  1460→      ["cote", values.odds],
  1461→  1461→    ];
  1462→  1462→    for (const [label, v] of criticals) {
  1463→  1463→      if (!v) warnings.push(`${label} non reconnu(e) — à remplir manuellement`);
  1464→  1464→    }
  1465→  1465→  }
  1466→  1466→
  1467→  1467→  /* ---- Ligne déduite de la sélection (aucun échec silencieux) ----
  1468→  1468→     « FC Porto -0,5 » → ligne « -0,5 » ; « Under 2,5 » → « 2,5 ».
  1469→  1469→     Seulement si aucune ligne n'a été trouvée ailleurs. */
  1470→  1470→  if (!values.line && values.selection) {
  1471→  1471→    const sel = values.selection.replace(/\s+/g, " ").trim();
  1472→  1472→    const sm =
  1473→  1473→      sel.match(/(?:^|\s)([+\-−–]\s?\d+(?:[.,]\d+)?)\s*$/) ??
  1474→  1474→      sel.match(/^(?:under|over|moins de|plus de|au-dessus de|en-dessous de)\s+(\d+(?:[.,]\d+)?)/i);
  1475→  1475→    if (sm) {
  1476→  1476→      values.line = (sm[1] ?? "").replace(/[−–—]/g, "-").replace(/\s+/g, "").slice(0, 60);
  1477→  1477→      if (values.line) {
  1478→  1478→        filled.push("ligne (dérivée de la sélection)");
  1479→  1479→        warnings.push("ligne déduite de la sélection — vérifiez-la");
  1480→  1480→      }
  1481→  1481→    }
  1482→  1482→  }
  1483→  1483→
  1484→  1484→  /* ---- Dernier repli pour l'affiche : motif « X vs Y » ----
  1485→  1485→     Recherche LIGNE PAR LIGNE (texte nettoyé) : évite qu'un préfixe de la
  1486→  1486→     ligne précédente (« EuroLeague », « Basketball »…) se colle à l'affiche
  1487→  1487→     sur un texte aplati. */
  1488→  1488→  if (!values.matchName) {
  1489→  1489→    const vsLine = cleanText(raw)
  1490→  1490→      .split("\n")
  1491→  1491→      .map((l) => stripInline(l).trim())
  1492→  1492→      .find((l) => /\bvs\.?\b/i.test(l));
  1493→  1493→    if (vsLine) {
  1494→  1494→      const vm2 = vsLine.match(/([A-Za-zÀ-ÿ0-9.''\- ]{3,48})\s+\bvs\.?\b\s+([A-Za-zÀ-ÿ0-9.''\- ]{3,48})/i);
  1495→  1495→      if (vm2) {
  1496→  1496→        values.matchName = `${vm2[1].trim()} vs ${vm2[2].trim()}`.slice(0, 160);
  1497→  1497→        filled.push("affiche (vs)");
  1498→  1498→      }
  1499→  1499→    }
  1500→  1500→  }
  1501→  1501→
  1502→  1502→  if (filled.length === 0) {
  1503→  1503→    parseErrors.push("aucun champ reconnu — vérifiez que le texte suit le format NODE.47 (sections A à G)");
  1504→  1504→  }
  1505→  1505→
  1506→  1506→  return { values, filled, warnings, parseErrors };
  1507→  1507→}
